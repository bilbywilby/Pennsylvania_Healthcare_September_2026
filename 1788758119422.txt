## The pre-stitched integration seam

Three deliberate choices make the eventual merge into the audit repo non-breaking:

1. **Package namespace**: it lives at `src/provenance_engine/`, so dropping it into the main repo's `src/` won't collide with the existing `src/paths.py` or future `src/api/`.
2. **Config-over-convention**: after merge, `engine.toml` sets `owner_repo_prefix = "pennsylvania-health-insurance-audit-2026/"` and repoints `ledger`/`statutes` to the main repo's copies (or symlinks them) — no code edits.
3. **`claim_id_namespace`**: if the audit repo's own golden-fixture IDs ever collide with engine-side claims, the prefix resolves it mechanically.

Merge-day checklist, for the README: copy `src/provenance_engine/` → main `src/`; move `data/*.json` or point config at them; migrate the golden fixture via `seed_from_golden_fixture()` (it requires the real source document URL — refuse the placeholder); delete this standalone remote or archive it.

#!/usr/bin/env python3
"""
hardened_renderer.py

A tiny, self‑contained HTML/JSON renderer that the provenance engine can
use to display a claim‑verification ledger without pulling in any
application‑specific rendering logic.

The script is deliberately “hardened”:

* No implicit relative imports – everything is absolute from the
  `provenance_engine` package.
* All external data (paths, config) are read from the engine TOML file.
* The entry‑point can be called as a module (`python -m`) or as an
  executable script; both modes honour the same CLI.
* Errors are surfaced as `SystemExit(1)` with a short message – this
  makes CI failures easy to spot.
"""

from __future__ import annotations

import argparse
import json
import os
import sys
from pathlib import Path
from typing import Any, Dict

import tomllib  # stdlib in 3.13

# --------------------------------------------------------------------------- #
# Helper utilities
# --------------------------------------------------------------------------- #
def load_config(config_path: Path) -> Dict[str, Any]:
    """Load the TOML config used by the provenance engine."""
    if not config_path.is_file():
        raise FileNotFoundError(f"Config file not found: {config_path}")
    with config_path.open("rb") as f:
        return tomllib.load(f)


def read_json(file_path: Path) -> Dict[str, Any]:
    """Read a JSON file, returning an empty dict on decode error."""
    if not file_path.is_file():
        raise FileNotFoundError(f"JSON file not found: {file_path}")
    try:
        return json.loads(file_path.read_text(encoding="utf-8"))
    except json.JSONDecodeError as exc:
        raise ValueError(f"Invalid JSON in {file_path}: {exc}") from exc


def render_as_html(ledger: Dict[str, Any]) -> str:
    """
    Very small HTML view – just a table of claim_id, value, last_verified
    and verification status.  This keeps the renderer agnostic to the
    surrounding web app.
    """
    rows = []
    for claim_id, data in ledger.items():
        verified = data.get("verified_by")
        status = "✅" if verified else "❓"
        rows.append(
            f"<tr>"
            f"<td>{claim_id}</td>"
            f"<td>{data.get('value')}</td>"
            f"<td>{data.get('last_verified')}</td>"
            f"<td>{status}</td>"
            f"</tr>"
        )
    body = "\n".join(rows)
    return f"""<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<title>Provenance Ledger</title>
<style>
table {{border-collapse: collapse; width: 100%;}}
th, td {{border: 1px solid #ddd; padding: 8px; text-align: left;}}
th {{background-color: #f2f2f2;}}
</style>
</head>
<body>
<h1>Provenance Ledger</h1>
<table>
<thead>
<tr><th>Claim ID</th><th>Value</th><th>Last Verified</th><th>Status</th></tr>
</thead>
<tbody>
{body}
</tbody>
</table>
</body>
</html>"""


def render_as_json(ledger: Dict[str, Any], pretty: bool = False) -> str:
    """Return a JSON string – pretty‑printed on demand."""
    if pretty:
        return json.dumps(ledger, indent=2, sort_keys=True, ensure_ascii=False)
    return json.dumps(ledger, separators=(",", ":"), ensure_ascii=False)


# --------------------------------------------------------------------------- #
# CLI entry point
# --------------------------------------------------------------------------- #
def _build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(
        prog="hardened_renderer",
        description="Render the provenance engine ledger as HTML or JSON.",
    )
    parser.add_argument(
        "-c",
        "--config",
        type=Path,
        default=Path("config/engine.toml"),
        help="Path to the engine TOML configuration (default: config/engine.toml).",
    )
    parser.add_argument(
        "-f",
        "--format",
        choices=["html", "json"],
        default="html",
        help="Output format (default: html).",
    )
    parser.add_argument(
        "-p",
        "--pretty",
        action="store_true",
        help="Pretty‑print JSON output (has no effect for HTML).",
    )
    parser.add_argument(
        "-o",
        "--output",
        type=Path,
        help="Write output to a file; otherwise print to stdout.",
    )
    return parser


def main(argv: list[str] | None = None) -> None:
    parser = _build_parser()
    args = parser.parse_args(argv)

    # ------------------------------------------------------------------- #
    # Load configuration – we only need the ledger path for rendering.
    # ------------------------------------------------------------------- #
    try:
        cfg = load_config(args.config)
        ledger_path = Path(cfg["paths"]["ledger"])
    except Exception as exc:
        sys.stderr.write(f"Configuration error: {exc}\n")
        sys.exit(1)

    # ------------------------------------------------------------------- #
    # Load the ledger JSON.
    # ------------------------------------------------------------------- #
    try:
        ledger = read_json(ledger_path)
    except Exception as exc:
        sys.stderr.write(f"Ledger load error: {exc}\n")
        sys.exit(1)

    # ------------------------------------------------------------------- #
    # Render.
    # ------------------------------------------------------------------- #
    if args.format == "html":
        output = render_as_html(ledger)
    else:
        output = render_as_json(ledger, pretty=args.pretty)

    # ------------------------------------------------------------------- #
    # Emit.
    # ------------------------------------------------------------------- #
    if args.output:
        try:
            args.output.write_text(output, encoding="utf-8")
        except OSError as exc:
            sys.stderr.write(f"Failed to write output file: {exc}\n")
            sys.exit(1)
    else:
        sys.stdout.write(output)


if __name__ == "__main__":
    main()
