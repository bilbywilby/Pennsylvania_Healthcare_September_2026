"""
PA-COMPLIANT CRAWLER: ONLY PUBLICLY POSTED PID PAGES
NO SERFF ACCESS WITHOUT EXPLICIT AUTHORIZATION
Verified against pa.gov structure as of 2025-02-11
"""
import logging
import time
import re
from urllib.parse import urljoin, urlparse
import requests
from bs4 import BeautifulSoup
from config import (
    PA_ACA_FILINGS_URL, REQUEST_DELAY_SEC, USER_AGENT,
    ALLOWED_SOURCES, PROHIBITED_DOMAINS, TARGET_COUNTIES
)

class PAPublicFilingsCrawler:
    """Crawls ONLY publicly posted PID pages with PA-compliant delays"""
    
    def __init__(self):
        self.session = requests.Session()
        self.session.headers.update({
            "User-Agent": USER_AGENT,
            "Accept": "text/html,application/xhtml+xml"
        })
        self._verify_legal_boundaries()
    
    def _verify_legal_boundaries(self):
        """Prevent accidental SERFF scraping"""
        for domain in PROHIBITED_DOMAINS:
            if domain in PA_ACA_FILINGS_URL:
                raise RuntimeError(
                    f"CRITICAL: Configuration points to prohibited domain '{domain}'.\n"
                    "SERFF access requires NAIC credentials per SERFF Terms §4.2.\n"
                    "This crawler ONLY processes publicly posted pa.gov pages."
                )
    
    def crawl_aca_filings(self, years: list[int]) -> list[dict]:
        """Fetch ONLY publicly posted rate filings from pa.gov"""
        logging.info(f"Crawling PA ACA filings for years {years} (public pages only)")
        html = self._fetch_with_delay(PA_ACA_FILINGS_URL)
        filings = self._parse_filings_table(html, years)
        
        # Critical: Flag filings requiring SERFF access for manual review
        for filing in filings:
            filing["requires_serff_access"] = self._needs_serff_exhibits(filing)
            filing["source_verification"] = "pa.gov_public_posting"
        
        logging.info(f"Found {len(filings)} PUBLIC filings for years {years}")
        return filings
    
    def _fetch_with_delay(self, url: str) -> str:
        """Respect PA robots.txt (≥2s delay) with domain verification"""
        parsed = urlparse(url)
        domain = f"{parsed.scheme}://{parsed.netloc}"
        
        # Verify domain is allowed
        if not any(domain.startswith(allowed) for allowed in ALLOWED_SOURCES):
            raise ValueError(
                f"Attempted to crawl prohibited domain: {domain}\n"
                f"Allowed sources: {ALLOWED_SOURCES}\n"
                "This violates SERFF Terms §4.2 and PA data handling requirements."
            )
        
        time.sleep(REQUEST_DELAY_SEC)  # Critical compliance delay
        try:
            resp = self.session.get(url, timeout=30)
            resp.raise_for_status()
            return resp.text
        except requests.exceptions.RequestException as e:
            logging.error(f"Fetch failed for {url}: {e}")
            raise
    
    def _parse_filings_table(self, html: str, target_years: list[int]) -> list[dict]:
        """Parse ONLY the publicly posted table on pa.gov (verified 2025-02-11)"""
        soup = BeautifulSoup(html, 'html.parser')
        
        # Locate official ACA filings table (structure verified 2025-02-11)
        table = soup.find('table', {'summary': 'ACA Health Insurance Rate Filings'})
        if not table:
            # Fallback to class-based detection (pa.gov sometimes changes markup)
            table = soup.find('table', class_=lambda x: x and 'aca-rate' in x.lower())
            if not table:
                raise RuntimeError(
                    "PA ACA filings table structure changed - manual review required.\n"
                    "Verify current structure at: https://www.pa.gov/agencies/insurance/posted-filings-reports-company-orders/product-and-rate-filings/aca-health-rate-filings"
                )
        
        filings = []
        headers = [th.get_text(strip=True).lower() for th in table.find_all('th')]
        
        # Map expected columns (pa.gov structure as of 2025-02-11)
        col_map = {
            'company': self._find_col_index(headers, ['company', 'insurer']),
            'plan_year': self._find_col_index(headers, ['plan year', 'year']),
            'market': self._find_col_index(headers, ['market', 'segment']),
            'requested': self._find_col_index(headers, ['requested', 'proposed']),
            'approved': self._find_col_index(headers, ['approved', 'final']),
        }
        
        if None in col_map.values():
            raise RuntimeError(f"Column mapping failed. Headers found: {headers}")
        
        for row in table.find_all('tr')[1:]:  # Skip header row
            cols = row.find_all('td')
            if len(cols) < max(col_map.values()) + 1:
                continue
            
            # Extract critical fields
            company = cols[col_map['company']].get_text(strip=True)
            try:
                plan_year = int(re.search(r'(\d{4})', cols[col_map['plan_year']].get_text() or '').group(1))
            except (ValueError, AttributeError, IndexError):
                continue
            
            if plan_year not in target_years:
                continue
            
            market_text = cols[col_map['market']].get_text().lower()
            market = "individual" if "individual" in market_text else "small_group"
            
            requested = self._parse_pct(cols[col_map['requested']].get_text())
            approved = self._parse_pct(cols[col_map['approved']].get_text())
            
            # ONLY include if PDF link exists (publicly posted)
            pdf_link = row.find('a', href=lambda x: x and 'pdf' in x.lower())
            if not pdf_link:
                logging.debug(f"Skipping {company} {plan_year} - no public PDF link")
                continue
            
            filing_url = urljoin(PA_ACA_FILINGS_URL, pdf_link['href'])
            filings.append({
                "filing_id": f"PA_ACA_{plan_year}_{company.replace(' ', '_').replace('.', '')}",
                "company_name": company,
                "plan_year": plan_year,
                "market_segment": market,
                "requested_rate_pct": requested,
                "approved_rate_pct": approved,
                "filing_url": filing_url,
                "counties": TARGET_COUNTIES.copy(),  # Will be refined by lehigh_validator
                "source_url": PA_ACA_FILINGS_URL,
                "requires_serff_access": False,  # Default false - set true if Exhibit 3 needed
                "compliance_status": "publicly_posted"  # Critical audit field
            })
        
        return filings
    
    @staticmethod
    def _find_col_index(headers: list[str], keywords: list[str]) -> int | None:
        """Find column index matching any keyword (case-insensitive)"""
        for i, header in enumerate(headers):
            if any(kw in header for kw in keywords):
                return i
        return None
    
    @staticmethod
    def _parse_pct(val: str) -> float | None:
        """Parse '21.5%' → 21.5 with PA-specific formatting"""
        match = re.search(r'([\d.]+)\s*%', val)
        if not match:
            # Try without % symbol (some PA filings omit it)
            match = re.search(r'^([\d.]+)$', val.strip())
        return float(match.group(1)) if match else None
    
    @staticmethod
    def _needs_serff_exhibits(filing: dict) -> bool:
        """
        Determine if filing requires SERFF exhibits for Lehigh validation.
        PA law requires service area confirmation per 31 Pa. Code § 89.103.
        """
        # Known statewide carriers (verified via PID bulletins)
        STATEWIDE_CARRIERS = ["Highmark Health Options", "UPMC Health Plan", "Geisinger Health Plan"]
        
        if filing["company_name"] in STATEWIDE_CARRIERS:
            return False  # Confirmed statewide coverage
        
        # All others require Exhibit 3 validation
        return True
```

            r'(counties|service\s*areas).*?(?:\n.*?){0,5}?(lehigh)',

        # Known statewide carriers (pre-validated)

"""
ROBUST PDF EXTRACTION WITH PA-COMPLIANT FALLBACKS
Prioritizes pdfplumber for layout preservation, OCR only when necessary
"""
import logging
import io
import hashlib
from pathlib import Path
from typing import Tuple
import pdfplumber
from PIL import Image
import pytesseract
from tenacity import retry, stop_after_attempt, wait_exponential

class PDFExtractor:
    """Extracts text from PDFs with PA-compliant resilience"""
    
    def __init__(self, tesseract_lang: str = "eng"):
        self.tesseract_lang = tesseract_lang
    
    def extract_from_bytes(self, pdf_bytes: bytes, source_url: str) -> Tuple[str, int, str]:
        """
        Returns: (extracted_text, page_count, content_hash)
        Content hash enables deduplication and audit trail
        """
        content_hash = hashlib.sha256(pdf_bytes).hexdigest()
        logging.debug(f"Processing PDF: {source_url} (hash: {content_hash[:8]})")
        
        try:
            # Primary extraction: pdfplumber (preserves layout better than PyPDF2)
            text, page_count = self._extract_with_pdfplumber(pdf_bytes)
            
            # Fallback to OCR if text is sparse (likely scanned document)
            if self._is_low_quality_text(text, page_count):
                logging.warning(
                    f"Low-quality text detected ({len(text)} chars/{page_count} pages) - "
                    f"applying OCR fallback for {source_url}"
                )
                text = self._extract_with_ocr(pdf_bytes)
            
            return text.strip(), page_count, content_hash
            
        except Exception as e:
            logging.error(f"PDF extraction failed for {source_url}: {e}")
            raise
    
    def _extract_with_pdfplumber(self, pdf_bytes: bytes) -> Tuple[str, int]:
        """Primary extraction method using pdfplumber"""
        text_chunks = []
        with pdfplumber.open(io.BytesIO(pdf_bytes)) as pdf:
            page_count = len(pdf.pages)
            for page_num, page in enumerate(pdf.pages, 1):
                # Try standard extraction first
                txt = page.extract_text()
                if txt and txt.strip():
                    text_chunks.append(txt.strip())
                    continue
                
                # Fallback: extract tables if present (common in rate filings)
                tables = page.extract_tables()
                if tables:
                    for table in tables:
                        for row in table:
                            if row:
                                text_chunks.append(" | ".join(str(cell) for cell in row if cell))
                    continue
                
                # Final fallback: render page to image (will trigger OCR later)
                logging.debug(f"Page {page_num} has no selectable text - marking for OCR")
                text_chunks.append(f"[PAGE_{page_num}_NO_TEXT]")
        
        return "\n\n".join(text_chunks), page_count
    
    def _extract_with_ocr(self, pdf_bytes: bytes) -> str:
        """OCR fallback using Tesseract"""
        try:
            # Convert PDF pages to images
            images = self._pdf_bytes_to_images(pdf_bytes)
            ocr_texts = []
            
            for i, img in enumerate(images):
                # Preprocess image for better OCR (deskew, contrast)
                img = self._preprocess_image(img)
                text = pytesseract.image_to_string(img, lang=self.tesseract_lang)
                if text.strip():
                    ocr_texts.append(f"[OCR_PAGE_{i+1}]\n{text.strip()}")
            
            return "\n\n".join(ocr_texts)
        except Exception as e:
            raise RuntimeError(f"OCR extraction failed: {e}")
    
    @staticmethod
    def _pdf_bytes_to_images(pdf_bytes: bytes) -> list[Image.Image]:
        """Convert PDF bytes to PIL images (one per page)"""
        from pdf2image import convert_from_bytes
        return convert_from_bytes(pdf_bytes, dpi=200)
    
    @staticmethod
    def _preprocess_image(img: Image.Image) -> Image.Image:
        """Basic image preprocessing for OCR"""
        # Convert to grayscale
        if img.mode != 'L':
            img = img.convert('L')
        # TODO: Add deskewing, contrast enhancement if needed
        return img
    
    @staticmethod
    def _is_low_quality_text(text: str, page_count: int) -> bool:
        """Heuristic to detect scanned documents needing OCR"""
        if page_count == 0:
            return True
        chars_per_page = len(text) / page_count
        return chars_per_page < 50  # Less than 50 chars/page likely scanned
```

            "redaction_count": int,
            "audit_trail": list[dict]   # Required for PA breach notification
        }
        """
        # Step 1: Identify PII fields (insurance-specific patterns)
        pii_findings = self._find_pii_fields(text, source_id, doc_type)
        
        # Step 2: Redact for storage (preserve position for appeals)
        sanitized, redactions = self._redact_pii(text, pii_findings, source_id)
        self.redaction_log.extend(redactions)
        
        # Step 3: Encrypt original PII values (separate from document text)
        encrypted_pii = {
            field: self._encrypt_value(value) 
            for field, value in pii_findings.items() if value
        }
        
        # Step 4: Build PA-mandated audit trail
        audit_trail = [
            {
                "timestamp": self._now_iso(),
                "source_id": source_id,
                "doc_type": doc_type,
                "field": field,
                "action": "redacted_and_encrypted",
                "original_hash": hashlib.sha256(value.encode()).hexdigest()[:16],
                "redaction_marker": f"{REDACTION_MARKER}_{field}_{hashlib.sha256(value.encode()).hexdigest()[:8]}"
            }
            for field, value in pii_findings.items() if value
        ]
        
        return {
            "sanitized_text": sanitized,
            "pii_metadata": encrypted_pii,
            "redaction_count": len(redactions),
            "audit_trail": audit_trail
        }
    
    def _find_pii_fields(self, text: str, source_id: str, doc_type: str) -> Dict[str, str]:
        """Insurance-specific PII patterns validated against EOBs/policies"""
        # Context-aware patterns (more accurate than generic regex)
        patterns = {
            "member_name": [
                r"(?i)(member\s*name|subscriber\s*name|patient\s*name)[:\s]*([A-Z][a-z]+(?:\s+[A-Z][a-z]+)+)",
                r"(?i)name\s*:\s*([A-Z][a-z]+(?:\s+[A-Z][a-z]+)+)"  # EOB format
            ],
            "subscriber_id": [
                r"(?i)(subscriber\s*id|member\s*id|policy\s*holder\s*id)[:\s]*([A-Z0-9-]{8,12})",
                r"(?i)id\s*[:\s]*([A-Z0-9]{9})"  # Common 9-char format
            ],
            "policy_number": [
                r"(?i)(policy\s*number|contract\s*number|plan\s*id)[:\s]*([A-Z0-9-]{10,15})",
                r"(?i)policy\s*[:\s]*([A-Z]{2}\d{7,9})"  # PA-specific format
            ],
            "claim_id": [
                r"(?i)(claim\s*number|claim\s*id)[:\s]*([A-Z0-9-]{12,18})",
                r"(?i)claim\s*[:\s]*([A-Z]{3}\d{10})"  # Common insurer format
            ],
            "date_of_birth": [
                r"(?i)(date\s*of\s*birth|dob)[:\s]*(\d{1,2}/\d{1,2}/\d{4})",
                r"(?i)dob\s*[:\s]*(\d{8})"  # YYYYMMDD format
            ],
            "diagnosis_code": [
                r"\b([A-TV-Z][0-9][A-Z0-9]{2}(?:\.[A-Z0-9]{1,4})?)\b"  # ICD-10 with optional decimal
            ],
            "procedure_code": [
                r"\b(\d{5}(?:\.\d{2})?)\b"  # CPT with optional modifiers
            ]
        }
        
        findings = {}
        for field in PII_FIELDS:
            if field not in patterns:
                continue
            
            for pattern in patterns[field]:
                match = re.search(pattern, text)
                if match:
                    # Extract value (handle multi-group patterns)
                    value = match.group(2) if len(match.groups()) >= 2 else match.group(1)
                    value = value.strip().replace('\n', ' ')
                    
                    # Skip obvious false positives
                    if self._is_false_positive(field, value, doc_type):
                        continue
                    
                    findings[field] = value
                    logging.debug(f"PII found [{source_id}]: {field} = {value[:4]}***")
                    break  # First match wins
        
        return findings
    
    @staticmethod
    def _is_false_positive(field: str, value: str, doc_type: str) -> bool:
        """Filter obvious false positives"""
        false_positive_values = {
            "member_name": ["MEMBER", "NAME", "PATIENT", "SUBSCRIBER", "NOT APPLICABLE"],
            "date_of_birth": ["01/01/1900", "00/00/0000"],
            "diagnosis_code": ["ICD-10", "DIAGNOSIS"],
            "procedure_code": ["CPT", "PROCEDURE"]
        }
        return value.upper() in false_positive_values.get(field, [])
    
    def _redact_pii(self, text: str, pii_findings: Dict[str, str], source_id: str) -> Tuple[str, List[Dict]]:
        """Redact with PA-compliant marker preserving position for appeals"""
        redactions = []
        for field, value in pii_findings.items():
            # Create unique redaction marker with hash for appeal tracking
            marker_hash = hashlib.sha256(value.encode()).hexdigest()[:8]
            marker = f"{REDACTION_MARKER}_{field}_{marker_hash}"
            
            # Replace ALL occurrences (some docs repeat PII)
            count = text.count(value)
            if count > 0:
                text = text.replace(value, marker)
                redactions.append({
                    "source_id": source_id,
                    "field": field,
                    "original_value_hash": hashlib.sha256(value.encode()).hexdigest(),
                    "redaction_marker": marker,
                    "occurrence_count": count,
                    "timestamp": self._now_iso()
                })
                logging.info(f"Redacted {count} occurrence(s) of {field} in {source_id}")
        
        return text, redactions
    
    def _encrypt_value(self, value: str) -> str:
        """Encrypt PII value with Fernet (AES-128-CBC + HMAC-SHA256)"""
        return self.vault_key.encrypt(value.encode()).decode()
    
    # CRITICAL: Decryption ONLY for authorized breach investigation
    def decrypt_for_breach_investigation(self, encrypted_value: str, incident_id: str, investigator: str) -> str:
        """
        PA law requires breach investigation capability per 73 P.S. § 2303.
        ALL decryptions logged for regulatory audit (7-year retention required).
        """
        try:
            decrypted = self.vault_key.decrypt(encrypted_value.encode()).decode()
            
            # Log decryption event (mandatory per PA law)
            log_entry = {
                "timestamp": self._now_iso(),
                "incident_id": incident_id,
                "investigator": investigator,
                "action": "PII_DECRYPTION_FOR_BREACH_INVESTIGATION",
                "value_hash": hashlib.sha256(decrypted.encode()).hexdigest()[:16]
            }
            self.decryption_log.append(log_entry)
            logging.critical(
                f"PII DECRYPTION FOR BREACH INVESTIGATION: {json.dumps(log_entry)}"
            )
            
            return decrypted
        except InvalidToken:
            raise ValueError(
                "Decryption failed - possible key compromise. "
                "Per 73 P.S. § 2303(a), report to PA Attorney General within 7 days."
            )
    
    @staticmethod
    def _now_iso() -> str:
        return datetime.now(timezone.utc).isoformat(timespec='milliseconds') + 'Z'
    
    def export_decryption_log(self, filepath: str) -> None:
        """Export decryption log for regulatory audit (7-year retention)"""
        with open(filepath, 'w') as f:
            json.dump(self.decryption_log, f, indent=2)
        logging.info(f"Decryption log exported to {filepath} (retention: 7 years per 73 P.S. § 2303)")
