```python
"""
Market Auditor Module.
Handles data ingestion and basic statistical analysis of the 2026 scenario.
"""
from typing import Dict, List
from config.settings import Config

class MarketAuditor:
    """Audits market data for the 2026 hypothetical scenario."""

    def __init__(self, rating_areas: Dict[int, List[str]] = None):
        self.rating_areas = rating_areas or Config.RATING_AREAS
        self.carriers_2026 = Config.CARRIERS_2026
        self.carriers_2027 = Config.CARRIERS_2027_REQUESTS
        self.disclaimer = Config.DISCLAIMER

    def calculate_weighted_average(self, rates: Dict[str, float]) -> float:
        """Calculates the weighted average rate increase."""
        if not rates:
            return 0.0
        # In a real scenario, weights would be based on enrollment numbers.
        # Here we assume equal weighting for the hypothetical scenario.
        return sum(rates.values()) / len(rates)

    def get_high_risk_carriers(self, threshold: float = 20.0) -> List[str]:
        """Identifies carriers with rate increases above the threshold."""
        return [carrier for carrier, rate in self.carriers_2026.items() if rate > threshold]

    def generate_summary_report(self) -> Dict:
        """Generates a summary of the audit findings."""
        avg_2026 = self.calculate_weighted_average(self.carriers_2026)
        avg_2027 = self.calculate_weighted_average(self.carriers_2027)
        high_risk = self.get_high_risk_carriers()

        return {
            "audit_id": Config.PROVENANCE_ID,
            "year": Config.AUDIT_YEAR,
            "disclaimer": self.disclaimer,
            "statistics": {
                "average_rate_increase_2026": round(avg_2026, 2),
                "average_rate_request_2027": round(avg_2027, 2),
                "carriers_above_20_percent": len(high_risk),
                "high_risk_carriers": high_risk
            },
            "rating_areas_count": len(self.rating_areas)
        }
```

```python
"""
Rate Validator Module.
Checks for logical consistency and data integrity in the hypothetical data.
"""
from typing import Dict, List
from config.settings import Config

class RateValidator:
    """Validates rate increases against hypothetical constraints."""

    def __init__(self, carriers_2026: Dict[str, float], carriers_2027: Dict[str, float]):
        self.carriers_2026 = carriers_2026
        self.carriers_2027 = carriers_2027
        self.errors: List[str] = []
        self.warnings: List[str] = []

    def validate_positive_rates(self) -> bool:
        """Ensures all rates are positive."""
        valid = True
        for carrier, rate in self.carriers_2026.items():
            if rate < 0:
                self.errors.append(f"Negative rate found for {carrier} in 2026.")
                valid = False
        for carrier, rate in self.carriers_2027.items():
            if rate < 0:
                self.errors.append(f"Negative rate found for {carrier} in 2027.")
                valid = False
        return valid

    def validate_consistency(self) -> bool:
        """Checks if 2027 requests are logically consistent with 2026 (heuristic)."""
        # In a real scenario, we might check if a carrier increased significantly in 2026 
        # but decreased in 2027 without a stated reason.
        for carrier in self.carriers_2026:
            if carrier in self.carriers_2027:
                if self.carriers_2027[carrier] < self.carriers_2026[carrier]:
                    self.warnings.append(f"{carrier} shows a decrease in 2027 request vs 2026 increase.")
        return len(self.errors) == 0

    def run_validation(self) -> Dict:
        """Runs all validation checks."""
        self.errors = []
        self.warnings = []
        
        self.validate_positive_rates()
        self.validate_consistency()

        return {
            "valid": len(self.errors) == 0,
            "errors": self.errors,
            "warnings": self.warnings
        }
```

```python
"""
Report Generator Module.
Formats the audit findings into a human-readable white paper format.
"""
from typing import Dict
from config.settings import Config

class ReportGenerator:
    """Generates formatted reports from audit data."""

    def __init__(self, summary: Dict, validation: Dict):
        self.summary = summary
        self.validation = validation

    def generate_text_report(self) -> str:
        """Generates a plain text white paper."""
        lines = [
            "=" * 80,
            f"PENNSYLVANIA PROVENANCE ENGINE REPORT",
            f"ID: {self.summary['audit_id']}",
            f"Year: {self.summary['year']}",
            "=" * 80,
            "",
            "DISCLAIMER:",
            self.summary['disclaimer'],
            "",
            "STATISTICAL SUMMARY:",
            f"- Average 2026 Rate Increase: {self.summary['statistics']['average_rate_increase_2026']}%",
            f"- Average 2027 Rate Request: {self.summary['statistics']['average_rate_request_2027']}%",
            f"- Carriers with >20% Increase: {self.summary['statistics']['carriers_above_20_percent']}",
            "",
            "HIGH RISK CARRIERS:",
            ", ".join(self.summary['statistics']['high_risk_carriers']) if self.summary['statistics']['high_risk_carriers'] else "None",
            "",
            "VALIDATION RESULTS:",
            f"- Status: {'PASSED' if self.validation['valid'] else 'FAILED'}",
        ]
        
        if self.validation['warnings']:
            lines.append("WARNINGS:")
            for w in self.validation['warnings']:
                lines.append(f"  - {w}")
        
        if self.validation['errors']:
            lines.append("ERRORS:")
            for e in self.validation['errors']:
                lines.append(f"  - {e}")

        lines.append("")
        lines.append("=" * 80)
        
        return "\n".join(lines)
```

```python
"""Unit tests for the Rate Validator."""
import pytest
from src.provenance_engine.validator import RateValidator

@pytest.fixture
def validator():
    return RateValidator(
        {"CarrierA": 10.0, "CarrierB": 20.0},
        {"CarrierA": 15.0, "CarrierB": 25.0}
    )

def test_validate_positive_rates(validator):
    """Test positive rate validation."""
    result = validator.run_validation()
    assert result['valid'] is True
    assert len(result['errors']) == 0

def test_validation_warnings(validator):
    """Test warning generation."""
    result = validator.run_validation()
    # CarrierA increased from 10 to 15, CarrierB from 20 to 25. No decrease.
    # This test checks if warnings are empty when logic holds.
    # We expect warnings only if 2027 < 2026.
    assert len(result['warnings']) == 0

def test_negative_rate_error():
    """Test negative rate detection."""
```

### How to Deploy
1.  **Create the directories** using your terminal:
    ```bash
    mkdir -p pa-provenance-engine/{config,src/provenance_engine,data,tests,.github/workflows}
    ```
2.  **Copy the files** above into their respective locations.
3.  **Install dependencies**:
    ```bash
    cd pa-provenance-engine
    pip install -r requirements.txt
    ```
4.  **Run tests**:
    ```bash
    pytest
    ```
5.  **Run the engine** (conceptual example):
    ```python
    # In a Python shell or script:
    from src.provenance_engine.auditor import MarketAuditor
    from src.provenance_engine.reporter import ReportGenerator
    
    auditor = MarketAuditor()
    summary = auditor.generate_summary_report()
    from src.provenance_engine.validator import RateValidator
    validator = RateValidator(auditor.carriers_2026, auditor.carriers_2027)
    validation = validator.run_validation()
    
    reporter = ReportGenerator(summary, validation)
    print(reporter.generate_text_report())
    ```

This setup provides a clean, modular, and testable foundation for your hypothetical audit project.
