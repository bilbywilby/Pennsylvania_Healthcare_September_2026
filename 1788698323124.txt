```bash
import sys
from pathlib import Path

audit_script = '''#!/usr/bin/env python3
"""
04_audit.py – Carrier-Level Filing Auditor
Validates approved_rates_2026.csv (carrier-level extensions, rating areas 1-9).

This audit reinforces the same data contracts as 05_ingest_carrier_extensions.py,
ensuring carrier names, rating area ranges, and decimal bounds are canonical.
"""

import csv
import sys
from pathlib import Path
from decimal import Decimal, InvalidOperation
from audit.contract_validator import load_canonical_crosswalk

# ============================================================================
# CONSTANTS
# ============================================================================
PA_FIPS = "42"
MIN_RATE_CHANGE = Decimal("-50.00") # Regulatory floor (conservative)
MAX_RATE_CHANGE = Decimal("+50.00") # Regulatory ceiling (conservative)
VALID_RATING_AREAS = {1, 2, 3, 4, 5, 6, 7, 8, 9}

# Known carriers (authoritative list from PID/SERFF)
CANONICAL_CARRIERS = {
"Ambetter from Buckeye Health Plan",
"Ambetter from Confusion Hill",
"Ambetter from Sunflower Health Plan",
"Ambetter from WellSense Health Plan",
"Asuransi / Affinity Health",
"Geisinger Health",
"Highmark Inc.",
"Highmark Benefits Group",
"Lemonade Insurance",
"Partners Insurance Co.",
"Summa Insurance",
"UnitedHealthcare Insurance Company",
}

# ============================================================================
# VALIDATORS
# ============================================================================

def validate_carrier_name(carrier: str, line_no: int) -> bool:
"""Verify carrier is in canonical list."""
if carrier not in CANONICAL_CARRIERS:
print(f"❌ [Line {line_no}] Carrier '{carrier}' not in canonical list.")
print(f" Valid: {sorted(CANONICAL_CARRIERS)}")
return False
return True


def validate_rating_areas(area_list_str: str, line_no: int) -> bool:
"""Parse semicolon-delimited rating areas; ensure all are in {1-9}."""
try:
areas = set(int(a.strip()) for a in area_list_str.split(";") if a.strip())
invalid = areas - VALID_RATING_AREAS
if invalid:
print(f"❌ [Line {line_no}] Invalid rating areas: {invalid}")
return False
if not areas:
print(f"❌ [Line {line_no}] Rating area list is empty.")
return False
return True
except ValueError as e:
print(f"❌ [Line {line_no}] Rating area parse error: {e}")
return False


def validate_rate_change(rate_str: str, line_no: int) -> bool:
"""Verify rate change is finite Decimal, within bounds, quantized to 2 places."""
try:
rate = Decimal(rate_str.strip())

# Check for special values
if rate.is_nan() or rate.is_infinite():
print(f"❌ [Line {line_no}] Rate '{rate_str}' is NaN or Infinity.")
return False

# Check bounds
if rate < MIN_RATE_CHANGE or rate > MAX_RATE_CHANGE:
print(f"❌ [Line {line_no}] Rate {rate} outside [{MIN_RATE_CHANGE}, {MAX_RATE_CHANGE}].")
return False

# Check quantization
quantized = rate.quantize(Decimal("0.01"))
if rate != quantized:
print(f"❌ [Line {line_no}] Rate {rate} not quantized to 2 decimal places.")
return False

return True
except (InvalidOperation, ValueError) as e:
print(f"❌ [Line {line_no}] Rate parse error: {e}")
return False


def validate_row(row: dict, line_no: int, fips_to_area: dict) -> bool:
"""Validate a single CSV row."""
required_fields = {"carrier", "rating_areas", "rate_change_percent", "effective_date"}

# Check required fields
if not all(k in row for k in required_fields):
missing = required_fields - set(row.keys())
print(f"❌ [Line {line_no}] Missing fields: {missing}")
return False

carrier = row["carrier"].strip()
rating_areas_str = row["rating_areas"].strip()
rate_str = row["rate_change_percent"].strip()
effective_date = row["effective_date"].strip()

# Validate each field
checks = [
validate_carrier_name(carrier, line_no),
validate_rating_areas(rating_areas_str, line_no),
validate_rate_change(rate_str, line_no),
]

# Validate effective_date (ISO 8601)
try:
from datetime import date
date.fromisoformat(effective_date)
except ValueError:
print(f"❌ [Line {line_no}] Invalid ISO date: '{effective_date}'")
checks.append(False)

return all(checks)


# ============================================================================
# MAIN
# ============================================================================

def main():
input_csv = Path("data/public/approved_rates_2026.csv")
crosswalk_json = Path("data/public/pa_county_fips_crosswalk.json")

if not input_csv.exists():
print(f"❌ ERROR: {input_csv} not found.")
sys.exit(1)

if not crosswalk_json.exists():
print(f"❌ ERROR: {crosswalk_json} not found.")
sys.exit(1)

# Load crosswalk
try:
fips_to_name, fips_to_area = load_canonical_crosswalk(crosswalk_json)
except Exception as e:
print(f"❌ ERROR loading crosswalk: {e}")
sys.exit(1)

# Audit CSV rows
errors = []
with open(input_csv, "r", encoding="utf-8") as f:
reader = csv.DictReader(f)
for line_no, row in enumerate(reader, start=2): # start=2 (header is line 1)
if not validate_row(row, line_no, fips_to_area):
errors.append(line_no)

if errors:
print(f"\\n❌ AUDIT FAILED: {len(errors)} row(s) failed validation: {errors}")
sys.exit(1)

print(f"✅ AUDIT PASSED: {input_csv.name}")
print(f"✅ Loaded {len(fips_to_name)} counties from {crosswalk_json.name}")
sys.exit(0)


if __name__ == "__main__":
main()
'''

# Write to audit/04_audit.py
audit_dir = Path("audit")
audit_dir.mkdir(parents=True, exist_ok=True)
audit_path = audit_dir / "04_audit.py"
audit_path.write_text(audit_script, encoding="utf-8")
audit_path.chmod(0o755)

print(f"✅ [WRITE] Refactored audit/04_audit.py (Carrier Filing Auditor)")
print(f"✅ Validates: carrier names, rating areas (1-9), rate bounds (-50% to +50%), ISO dates")

EOF
```
