which is the reason this module exists at all.

Provenance: this table matches two independently-authored prior versions
(bootstrap/02_crosswalk.py's original RATING_AREAS/COUNTY_FIPS dicts and
scripts/bootstrap_data.py's COUNTIES_RAW string) that agreed on all 67
counties, and was spot-checked against real PA Insurance Department 2026
rate-filing rating-area listings (e.g. Highmark Coverage Advantage: areas
1 and 4; Geisinger Health Plan: areas 2,3,5,6,7,9). A third candidate table
was checked against this one and rejected — it disagreed on 57 of 67 rows
and had a corrupted FIPS/name pairing (see conversation history for that
comparison) — so treat any future proposed replacement of this table the
same way: diff it against this one and against real filings before
accepting it, not on the strength of looking well-formatted.
"""

# Fail loudly and immediately on import if the table is ever bad, rather
# than silently deduping-then-asserting-a-count — a count check alone
# can't tell a corrupted substitution from a correct table (a wrong FIPS
# swapped in for a missing one keeps the total at 67 either way).
_fips_seen = set()
_names_seen = set()
for _entry in PA_COUNTIES_CANONICAL:
    if _entry["fips"] in _fips_seen:
        raise AssertionError(f"Duplicate FIPS in PA_COUNTIES_CANONICAL: {_entry['fips']}")
    if _entry["county"] in _names_seen:
        raise AssertionError(f"Duplicate county name in PA_COUNTIES_CANONICAL: {_entry['county']}")
    if not (1 <= _entry["rating_area"] <= 9):
        raise AssertionError(f"Rating area out of range for {_entry['county']}: {_entry['rating_area']}")
    _fips_seen.add(_entry["fips"])
    _names_seen.add(_entry["county"])
if len(PA_COUNTIES_CANONICAL) != 67:
    raise AssertionError(f"Expected 67 PA counties, got {len(PA_COUNTIES_CANONICAL)}")


def get_flat_crosswalk_dict() -> Dict:
    """Flat-list shape: {"state_fips": "42", "state_abbr": "PA", "counties": [...]}."""
    return {
        "state_fips": "42",
        "state_abbr": "PA",
        "counties": [
            {"fips": e["fips"], "canonical_name": e["county"], "rating_area": e["rating_area"]}
            for e in PA_COUNTIES_CANONICAL
        ],
    }


def get_area_keyed_crosswalk_dict() -> Dict[str, List[Dict[str, str]]]:
    """Area-keyed shape: {"1": [{"county": "...", "fips": "..."}], "2": [...], ...}."""
    area_map: Dict[str, List[Dict[str, str]]] = {str(i): [] for i in range(1, 10)}
    for e in PA_COUNTIES_CANONICAL:
        area_map[str(e["rating_area"])].append({"county": e["county"], "fips": e["fips"]})
    for area_id in area_map:
        area_map[area_id].sort(key=lambda x: x["county"])
    return area_map


def get_counties_by_area() -> Dict[int, List[str]]:
    """{1: ["Clarion", "Crawford", ...], 2: [...], ...} — for human-readable
    reports (e.g. scripts/bootstrap_data.py's rating_areas_2026.csv) that
    need "which counties are in area N" without re-deriving it themselves."""
    result: Dict[int, List[str]] = {i: [] for i in range(1, 10)}
    for e in PA_COUNTIES_CANONICAL:
        result[e["rating_area"]].append(e["county"])
    for area_id in result:
        result[area_id].sort()
    return result
