```python
def compute_rating_area_exposure(filings: List[Dict[str, Any]]) -> Dict[int, Dict[str, Any]]:
    """Covered-lives weighted rate change per rating area; unweighted fallback + warning."""
    exposure: Dict[int, Dict[str, Any]] = {}
    for area in range(1, 10):
        covering = [f for f in filings if area in f["rating_areas"]]
        if not covering:
            exposure[area] = {"weighted_rate_change": None, "basis": "no_active_carrier"}
            continue
        rows = [(f["rate_change"], f.get("lives_by_area", {}).get(area, 0.0)) for f in covering]
        total_lives = sum(l for _, l in rows)
        if total_lives > 0:
            weighted = sum(r * l for r, l in rows) / total_lives
            basis = "covered_lives"
        else:
            logger.warning("No covered-lives data for RA %d — falling back to unweighted mean.", area)
            weighted = sum(r for r, _ in rows) / len(rows)
            basis = "unweighted_fallback"
        exposure[area] = {
            "weighted_rate_change": round(weighted, 2),
            "carrier_count": len(covering),
            "total_covered_lives": int(total_lives),
            "basis": basis,
        }
    return exposure
```
