#### 1. `crawler/pennie_coverage.py` (NEW)
```python
"""
PENNIE / PID COVERAGE AREAS CRAWLER
Parses public 2026 county listings from PID (no SERFF, no RTKL needed)
"""
import logging
import re
from urllib.parse import urljoin
import requests
from bs4 import BeautifulSoup
from config import REQUEST_DELAY_SEC, USER_AGENT

class PennieCoverageParser:
    """Parses PID coverage-areas pages (individual + small group)"""
    
    BASE_URL = "https://www.pa.gov/agencies/insurance/posted-filings-reports-company-orders/product-and-rate-filings/aca-health-rate-filings"
    INDIVIDUAL_URL = BASE_URL + "/coverage-areas-individual-market"
    SMALL_GROUP_URL = BASE_URL + "/coverage-areas-small-group-market"  # exists per earlier search

    def __init__(self):
        self.session = requests.Session()
        self.session.headers.update({"User-Agent": USER_AGENT})

    def get_coverage_dict(self, market: str = "individual") -> dict[str, list[str]]:
        """Returns {carrier_name_normalized: [counties...]}"""
        url = self.INDIVIDUAL_URL if market == "individual" else self.SMALL_GROUP_URL
        time.sleep(REQUEST_DELAY_SEC)
        html = self.session.get(url, timeout=30).text
        soup = BeautifulSoup(html, "html.parser")

        coverage = {}
        # Each carrier is a heading + paragraph with county list
        sections = soup.find_all(["h2", "h3", "p"])
        current_carrier = None

        for el in sections:
            text = el.get_text(strip=True)
            if el.name in ("h2", "h3") and any(kw in text.lower() for kw in ["inc", "health plan", "assurance", "partners"]):
                current_carrier = self._normalize_carrier(text)
            elif current_carrier and "covers the following" in text.lower():
                counties = re.findall(r'([A-Za-z]+(?:\s+[A-Za-z]+)*)', text.split(":", 1)[-1])
                counties = [c.strip() for c in counties if c.strip() and c not in ("Pennsylvania", "counties")]
                coverage[current_carrier] = counties

        logging.info(f"Parsed {len(coverage)} carriers from {market} coverage page")
        return coverage

    @staticmethod
    def _normalize_carrier(name: str) -> str:
        return re.sub(r'\s+', ' ', name).strip().lower().replace("inc.", "inc").replace("llc", "")
```

#### 2. `processor/pennie_affordability.py` (NEW)
```python
"""
PARSes PENNIE PUBLIC AFFORDABILITY PDFs
Premium changes + terminations by county (2026)
"""
import logging
from pathlib import Path
import pdfplumber
import requests
import time
from config import REQUEST_DELAY_SEC

class PennieAffordabilityParser:
    PDF_URLS = {
        "premium_pct": "https://pennie.com/wp-content/uploads/2026/02/Net-Premium-Change-in-Dollar-Percentage-for-Pre-Renewal-Run-02.pdf",
        "premium_dollar": "https://pennie.com/wp-content/uploads/2026/02/Net-Premium-Change-in-Dollar-Percentage-for-Pre-Renewal-Run-01.pdf",
        "terminations": "https://pennie.com/wp-content/uploads/2026/02/Open-Enrollment-Comparison-2021-2026-43-Terminations-by-County-OE-2026-1.pdf"
    }

    def fetch_and_parse(self) -> dict:
        data = {}
        for key, url in self.PDF_URLS.items():
            time.sleep(REQUEST_DELAY_SEC)
            resp = requests.get(url, timeout=30)
            resp.raise_for_status()
            with pdfplumber.open(io.BytesIO(resp.content)) as pdf:
                text = "\n".join(page.extract_text() or "" for page in pdf.pages)
                data[key] = self._extract_county_values(text, key)
        return data

    @staticmethod
    def _extract_county_values(text: str, key: str) -> dict[str, float]:
        # Heuristic – PDFs contain lines like "Lehigh County: 205%" or tables
        county_map = {}
        lines = text.split("\n")
        for line in lines:
            match = re.search(r'([A-Za-z]+(?:\s+[A-Za-z]+)*)\s*[:\-]?\s*([\d.]+)', line)
            if match:
                county = match.group(1).strip().title()
                val = float(match.group(2))
                if "lehigh" in county.lower():
                    logging.info(f"Pennie Lehigh {key}: {val}")
                county_map[county] = val
        return county_map
```

#### 4. `crawler/lehigh_validator.py` – Replace `validate_from_filing_text` with coverage-page version
```python
def validate_from_coverage_pages(self, company: str, plan_year: int = 2026) -> dict:
    parser = PennieCoverageParser()
    cov_ind = parser.get_coverage_dict("individual")
    cov_sg = parser.get_coverage_dict("small_group")
    
    normalized = self._normalize_carrier(company)
    counties = set(cov_ind.get(normalized, []) + cov_sg.get(normalized, []))
    
    is_served = "Lehigh" in counties
    return {
        "is_lehigh_served": is_served,
        "confidence": "high",
        "evidence_snippets": [f"PID coverage page: {len(counties)} counties including Lehigh"] if is_served else ["Not listed on PID coverage page"],
        "requires_rtkl_request": False  # Almost never needed now
    }
```
