                            <div 
                              className={`h-full rounded-full ${rateVal > 25 ? 'bg-rose-500' : rateVal > 15 ? 'bg-amber-500' : 'bg-blue-500'}`} 
                              style={{ width: `${barWidthPct}%` }}
                            ></div>
                          )}
                        </div>
                      )}

                      <div className="text-xs text-slate-400 pt-1 flex justify-between items-center">
                        <span>{carrier.notes}</span>
                      </div>
                    </div>
                  );
                })}
              </div>
            </div>
          </div>
        )}

        {}
        {activeTab === 'attrition' && (
          <div className="space-y-8 animate-fadeIn">
            {/* Attrition Hero Stats */}
            <div className="grid grid-cols-1 md:grid-cols-3 gap-6">
              <div className="bg-slate-800/50 border border-slate-700/50 rounded-2xl p-6">
                <span className="text-xs font-semibold text-slate-400">Net Enrollment Contraction</span>
                <div className="text-3xl font-extrabold text-rose-400 mt-2">-63,131</div>
                <p className="text-xs text-slate-400 mt-1">Active Pennie consumers lost between Feb 1 & Aug 1, 2026 (-13.0%)</p>
              </div>

              <div className="bg-slate-800/50 border border-slate-700/50 rounded-2xl p-6">
                <span className="text-xs font-semibold text-slate-400">OE Non-Effectuations</span>
                <div className="text-3xl font-extrabold text-amber-400 mt-2">85,000</div>
                <p className="text-xs text-slate-400 mt-1">Consumers who selected plans but failed to pay initial binder</p>
              </div>

              <div className="bg-slate-800/50 border border-slate-700/50 rounded-2xl p-6">
                <span className="text-xs font-semibold text-slate-400">Cumulative Cancellations</span>
                <div className="text-3xl font-extrabold text-slate-200 mt-2">&gt;191,000</div>
                <p className="text-xs text-slate-400 mt-1">Aggregate cancellations including mid-year drops (108,000)</p>
              </div>
            </div>

            {/* Custom SVG Line Chart for Pennie Attrition */}
            <div className="bg-slate-800/40 border border-slate-700/50 rounded-2xl p-6 space-y-6">
              <div>
                <h3 className="text-base font-bold text-white">Pennie Active Enrollment Decay (Plan Year 2026)</h3>
                <p className="text-xs text-slate-400">Tracking post-subsidy expiration active membership drops across 2026</p>
              </div>

              {/* Chart Visual Component */}
              <div className="relative h-64 w-full bg-slate-900/80 rounded-xl p-4 flex flex-col justify-between border border-slate-700/40">
                <div className="absolute inset-0 p-6 flex flex-col justify-between pointer-events-none opacity-20">
                  <div className="border-b border-slate-600 w-full"></div>
                  <div className="border-b border-slate-600 w-full"></div>
                  <div className="border-b border-slate-600 w-full"></div>
                </div>

                {/* SVG Visual Bars / Line overlay */}
                <div className="relative z-10 h-full flex items-end justify-around pt-6 pb-2">
                  {ENROLLMENT_ATTRITION.map((pt, i) => {
                    const heightPct = ((pt.count - 400000) / (490000 - 400000)) * 100;
                    return (
                      <div key={i} className="flex flex-col items-center gap-2 group w-1/6">
                        <div className="opacity-0 group-hover:opacity-100 transition-all text-xs bg-slate-800 text-white px-2 py-1 rounded shadow border border-slate-700 mb-1">
                          {pt.count.toLocaleString()}
                        </div>
                        <div className="w-12 bg-slate-800 rounded-t-lg relative overflow-hidden flex items-end" style={{ height: '140px' }}>
                          <div 
                            className="w-full bg-gradient-to-t from-blue-600 to-rose-500 rounded-t-lg transition-all duration-500" 
                            style={{ height: `${heightPct}%` }}
                          ></div>
                        </div>
                        <span className="text-xs font-semibold text-slate-300 mt-2">{pt.date}</span>
                        <span className="text-[10px] text-slate-500">{pt.count.toLocaleString()}</span>
                      </div>
                    );
                  })}
                </div>
              </div>

              {/* Table Data View */}
              <div className="overflow-x-auto">
                <table className="w-full text-left text-xs text-slate-300">
                  <thead className="bg-slate-900/60 text-slate-400 uppercase font-semibold">
                    <tr>
                      <th className="p-3">Measurement Date</th>
                      <th className="p-3">Active Enrollees</th>
                      <th className="p-3">Net Change from Feb 1</th>
                      <th className="p-3">Percent Change (%)</th>
                      <th className="p-3">Key Event / Context</th>
                    </tr>
                  </thead>
                  <tbody className="divide-y divide-slate-800">
                    {ENROLLMENT_ATTRITION.map((row, idx) => (
                      <tr key={idx} className="hover:bg-slate-800/30">
                        <td className="p-3 font-medium text-white">{row.date}</td>
                        <td className="p-3 text-slate-200 font-mono">{row.count.toLocaleString()}</td>
                        <td className="p-3 text-rose-400 font-mono">{row.drop > 0 ? `-${row.drop.toLocaleString()}` : '—'}</td>
                        <td className="p-3 text-rose-400 font-mono">{row.drop > 0 ? `-${((row.drop / 486000) * 100).toFixed(2)}%` : '—'}</td>
                        <td className="p-3 text-slate-400">{row.label}</td>
                      </tr>
                    ))}
                  </tbody>
                </table>
              </div>
            </div>
          </div>
        )}

        {}
        {activeTab === 'timeline' && (
          <div className="space-y-6 animate-fadeIn">
            {/* Category Filter */}
            <div className="flex flex-wrap gap-2">
              {['All', 'Subsidy Policy', 'Regulatory', 'Litigation', 'Enrollment'].map((cat) => (
                <button
                  key={cat}
                  onClick={() => setSelectedTimelineCategory(cat)}
                  className={`px-3 py-1.5 rounded-xl text-xs font-semibold transition-all ${
                    selectedTimelineCategory === cat
                      ? 'bg-blue-600 text-white shadow-md'
                      : 'bg-slate-800 text-slate-400 hover:bg-slate-700'
                  }`}
                >
                  {cat}
                </button>
              ))}
            </div>

            {/* Timeline Vertical Path */}
            <div className="bg-slate-800/40 border border-slate-700/50 rounded-2xl p-6">
              <div className="relative border-l-2 border-slate-700 ml-4 pl-6 space-y-8">
                {TIMELINE_EVENTS
                  .filter(e => selectedTimelineCategory === 'All' || e.category === selectedTimelineCategory)
                  .map((evt, idx) => (
                    <div key={idx} className="relative group">
                      {/* Timeline Dot */}
                      <div className="absolute -left-[31px] top-1.5 w-4 h-4 rounded-full bg-slate-900 border-2 border-blue-500 group-hover:scale-125 group-hover:bg-blue-500 transition-all"></div>

                      <div className="bg-slate-900/60 border border-slate-700/40 rounded-xl p-4 space-y-2 hover:border-slate-600 transition-all">
                        <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-1">
                          <span className="text-xs font-mono text-blue-400 font-bold">{evt.date}</span>
                          <span className="text-[10px] uppercase tracking-wider font-bold px-2 py-0.5 bg-slate-800 text-slate-300 rounded border border-slate-700">
                            {evt.category}
                          </span>
                        </div>
                        <h4 className="text-sm font-bold text-white">{evt.title}</h4>
                        <p className="text-xs text-slate-300 leading-relaxed">{evt.desc}</p>
                      </div>
                    </div>
                  ))}
              </div>
            </div>
          </div>
        )}

        {}
        {activeTab === 'calculator' && (
          <div className="max-w-3xl mx-auto space-y-6 animate-fadeIn">
            <div className="bg-slate-800/50 border border-slate-700/50 rounded-2xl p-6 space-y-6">
              <div>
                <h3 className="text-lg font-bold text-white flex items-center gap-2">
                  <IconCalculator />
                  <span>Personalized 2027 Rate Shock & Mapping Estimator</span>
                </h3>
                <p className="text-xs text-slate-400 mt-1">Estimate potential monthly premium adjustments under proposed 2027 filings</p>
              </div>

              <div className="space-y-4">
                {/* Inputs */}
                <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                  <div>
                    <label className="block text-xs font-semibold text-slate-300 mb-1">Select Carrier</label>
                    <select
                      value={calcCarrier}
                      onChange={(e) => setCalcCarrier(e.target.value)}
                      className="w-full bg-slate-900 border border-slate-700 text-white text-xs rounded-xl p-3 focus:outline-none focus:border-blue-500"
                    >
                      {INDIVIDUAL_CARRIERS.map((c, i) => (
                        <option key={i} value={c.name}>{c.name} (+{c.rate}%)</option>
                      ))}
                    </select>
                  </div>

                  <div>
                    <label className="block text-xs font-semibold text-slate-300 mb-1">Current Metal Tier</label>
                    <select
                      value={calcTier}
                      onChange={(e) => setCalcTier(e.target.value)}
                      className="w-full bg-slate-900 border border-slate-700 text-white text-xs rounded-xl p-3 focus:outline-none focus:border-blue-500"
                    >
                      <option value="bronze">Bronze Tier (Subject to Ambetter Plan Mapping)</option>
                      <option value="silver">Silver Tier</option>
                      <option value="gold">Gold Tier</option>
                    </select>
                  </div>
                </div>

                <div>
                  <label className="block text-xs font-semibold text-slate-300 mb-1">Current Monthly Premium ($)</label>
                  <input
                    type="number"
                    value={calcCurrentPremium}
                    onChange={(e) => setCalcCurrentPremium(Number(e.target.value))}
                    className="w-full bg-slate-900 border border-slate-700 text-white text-xs rounded-xl p-3 focus:outline-none focus:border-blue-500"
                  />
                </div>
              </div>

              {/* Calculation Output Box */}
              <div className="bg-slate-900/80 border border-blue-500/30 rounded-2xl p-6 space-y-4">
                <span className="text-xs font-bold text-blue-400 uppercase tracking-wider">Estimated Premium Output</span>
                
                <div className="grid grid-cols-1 sm:grid-cols-3 gap-4 text-center">
                  <div className="p-3 bg-slate-800/60 rounded-xl">
                    <span className="text-[10px] text-slate-400">Effective Rate Hike</span>
                    <div className="text-xl font-bold text-amber-400">+{calculatorResults.netIncreasePct.toFixed(2)}%</div>
                  </div>

                  <div className="p-3 bg-slate-800/60 rounded-xl">
                    <span className="text-[10px] text-slate-400">Est. New Monthly Cost</span>
                    <div className="text-xl font-bold text-white">${calculatorResults.estimatedNewPremium.toFixed(2)}</div>
                  </div>

                  <div className="p-3 bg-slate-800/60 rounded-xl">
                    <span className="text-[10px] text-slate-400">Monthly Increase</span>
                    <div className="text-xl font-bold text-rose-400">+${calculatorResults.monthlyDifference.toFixed(2)}</div>
                  </div>
                </div>

                <div className="p-3 bg-blue-950/40 border border-blue-800/40 rounded-xl text-xs text-blue-200">
                  <strong>Mechanics Note:</strong> {calculatorResults.note}
                </div>
              </div>
            </div>
          </div>
        )}
      </main>

      {}
      <footer className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 mt-12 pt-6 border-t border-slate-800 text-center text-xs text-slate-500">
        <p>Pennsylvania 2027 Health Insurance Rate Filings & Marketplace Analytics Portal.</p>
        <p className="mt-1">Data compiled from Pennsylvania Insurance Department (PID), Pennie Public Dashboards & Federal CMS filings.</p>
      </footer>
    </div>
  );
}
