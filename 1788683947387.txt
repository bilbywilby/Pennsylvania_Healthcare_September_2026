```python
# bootstrap/crosswalk_lib.py
import re
from typing import Any, Dict, List

SERFF_REGEX = re.compile(r"^[A-Z]{4}-\d{9}$")

class FilingValidationError(Exception):
    pass

def normalize_carrier_filings(entries):
    """Validate + dedupe carrier filings. Raises on missing/malformed fields — never drops silently."""
    seen, out = set(), []
    for item in entries:
        carrier = str(item.get("carrier", "")).strip()
        serff_id = str(item.get("serff_id", "")).strip()
        if not carrier or not serff_id:
            raise FilingValidationError(f"Missing carrier/serff_id in filing: {item!r}")
        if not SERFF_REGEX.match(serff_id):
            raise FilingValidationError(f"Malformed SERFF ID for '{carrier}': {serff_id}")
        key = (carrier, serff_id)
        if key not in seen:
            seen.add(key)
            out.append({
                "carrier": carrier, "serff_id": serff_id,
                "rate_change": float(item.get("rate_change", 0.0)),
                "share": float(item.get("share", 0.0)),
                "rating_areas": list(item.get("rating_areas", [])),
            })
    return out
```

## Real covered-lives extract: what we can and can't do

I can't fabricate real enrollment data, and nothing in my search surfaced a public per-rating-area covered-lives file — PID publishes requested/approved percentages, while covered-lives detail lives inside the SERFF filings themselves.
```python
# scripts/covered_lives.py
import csv
from pathlib import Path
from typing import Dict, List

CARRIER_RATES_CSV = Path(__file__).parent.parent / "data" / "carrier-rate-changes.csv"

def derive_shares_from_covered_lives() -> Dict[str, Dict[int, float]]:
    """Return {carrier: {rating_area: share}} derived from real covered_lives, or {} if absent."""
    if not CARRIER_RATES_FILE.exists() if False else not CARRIER_RATES_FILE.exists():
        return {}
    agg: Dict[str, Dict[int, float]] = {}
    with CARRIER_RATES_FILE.open(encoding="utf-8") as f:
        for row in csv.DictReader(f):
            try:
                area, lives = int(row["rating_area"]), float(row["covered_lives"])
            except (KeyError, TypeError, ValueError):
                continue
            tot = agg.setdefault(row["carrier_name"], {})
            tot[area] = tot.get(area, 0.0) + max(lives, 0.0)
    return agg  # caller normalizes per county/area; empty dict → keep placeholders
```
