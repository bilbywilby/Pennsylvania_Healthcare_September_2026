tests/test_provenance.py (core gates — these encode the standards, not test trivia):

import json, tempfile
from datetime import date, timedelta
from pathlib import Path
import pytest
from provenance_engine.provenance import (
    Claim, ProvenanceLedger, InvalidLedgerError, UnknownClaimError, ArtifactValidator,
)

def make_claim(**over):
    base = dict(
        claim_id="test_claim", value=42,
        source_citation="PID Annual Report 2026, p.14",
        source_url="https://www.insurance.pa.gov/",
        last_verified=date.today(), verified_by="test",
    )
    base.update(over)
    return Claim(**base)

class TestLedgerIntegrity:
    def test_placeholder_url_rejected(self):
        with pytest.raises(Exception):
            make_claim(source_url="https://example.com/report")

    def test_missing_local_path_rejected(self):
        with pytest.raises(Exception):
            make_claim(source_url="docs/nonexistent.pdf")

    def test_key_claim_id_mismatch_rejected(self, tmp_path):
        bad = {"wrong_key": make_claim().model_dump(mode="json")}
        p = tmp_path / "ledger.json"
        p.write_text(json.dumps(bad))
        with pytest.raises(InvalidLedgerError):
            ProvenanceLedger.load(p)

    def test_register_refuses_silent_overwrite(self):
        ledger = ProvenanceLedger({"test_claim": make_claim()})
        with pytest.raises(Exception):
            ledger.register(make_claim(value=99))  # same id, no overwrite flag

class TestValidator:
    def _ledger(self):
        return ProvenanceLedger({
            "fresh_claim": make_claim(claim_id="fresh_claim"),
            "stale_claim": make_claim(claim_id="stale_claim",
                                      last_verified=date.today() - timedelta(days=365)),
        })

    def test_unknown_claim_reported(self):
        v = ArtifactValidator(self._ledger())
        assert any("unknown_or_typo" in s for s in v.validate_artifact(
            {"data": {"claim_id": "unknown_or_typo"}}))

    def test_stale_claim_reported(self):
        v = ArtifactValidator(self._ledger())
        assert any("STALE" in s for s in v.validate_artifact(
            {"data": {"claim_id": "stale_claim"}}))

    def test_verified_artifact_clean(self):
        v = ArtifactValidator(self._ledger())
        assert v.validate_artifact({"data": {"claim_id": "fresh_claim"}}) == []

    def test_unverified_statute_fails_closed(self, tmp_path):
        v = ArtifactValidator(self._ledger(),
                               statutes_path=tmp_path / "nonexistent.json")
        assert any("verified column" in s for s in v.validate_artifact(
            {"statute": "Sec. 28-725"}))

    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - run: python -m provenance_engine.provenance --config config/engine.toml
        continue-on-error: false

(The final step will fail until the ledger is seeded with real claims — that's the fail-closed design working, and it documents itself in the Actions log.)

The pre-stitched integration seam
Three deliberate choices make the eventual merge into the audit repo non-breaking:

Package namespace: it lives at src/provenance_engine/, so dropping it into the main repo's src/ won't collide with the existing src/paths.py or future src/api/.
Config-over-convention: after merge, engine.toml sets owner_repo_prefix = "pennsylvania-health-insurance-audit-2026/" and repoints ledger/statutes to the main repo's copies (or symlinks them) — no code edits.
claim_id_namespace: if the audit repo's own golden-fixture IDs ever collide with engine-side claims, the prefix resolves it mechanically.
Merge-day checklist, for the README: copy src/provenance_engine/ → main src/; move data/*.json or point config at them; migrate the golden fixture via seed_from_golden_fixture() (it requires the real source document URL — refuse the placeholder); delete this standalone remote or archive it.
