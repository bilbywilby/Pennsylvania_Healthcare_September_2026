// Provenance-gated policy store — replaces dashboard's unsourced
// policyDerivation constants. Values arrive exclusively from
// /metrics/summary, whose payload cites PHIEA (2026a) / PID (2026a).

interface Act54Impact {
  restored_enrollees: number;
  premium_reduction_low_pct: number;
  premium_reduction_high_pct: number;
  enrollees_with_reduced_costs: number;
  appropriated_as_of_may_2026_usd: number;
  ep_tc_lost_annual_usd: number;
}

interface SummaryMetrics {
  federal_eptc_value_lost_annual_usd: number;
  act54_pennie_published_50m_impact: Act54Impact;
  roi_note: string;
}

// PROVENANCE: linear scaling below is an ASSUMPTION, not a Pennie-published
// relationship. Pennie publishes only the $50M impact point (44,000
// restored, 9-12% reduction, 280k+ helped — PHIEA 2026a). Any UI consuming
// scaled output MUST display the sourcing label or the number is not shippable.
export interface ScaledAct54Projection {
  projectedRestoredEnrollees: number;
  projectedEnrolleesWithReducedCosts: number;
  premiumReductionRange: [number, number];
  fundingShortfallVsEPTC: number;
  provenanceLabel: "PENNIE_PUBLISHED_POINT" | "ASSUMED_LINEAR_SCALING";
}

export const fetchVerifiedPolicyData = async (): Promise<SummaryMetrics> => {
  const response = await fetch('http://localhost:8000/metrics/summary');
  if (!response.ok) {
    throw new Error(`API returned status: ${response.status}`);
  }
  return response.json();
};

export const applyAct54RestorationModel = (
  metrics: SummaryMetrics,
  fundingLevelUsd: number
): ScaledAct54Projection => {
  const baselineImpact = metrics.act54_pennie_published_50m_impact;
  const atBaseline = fundingLevelUsd === 50000000;

  return {
    projectedRestoredEnrollees: Math.round(
      baselineImpact.restored_enrollees * (fundingLevelUsd / 50000000)
    ),
    projectedEnrolleesWithReducedCosts: Math.round(
      baselineImpact.enrollees_with_reduced_costs * (fundingLevelUsd / 50000000)
    ),
    // Percent bounds held static — no verified non-linear scaling exists.
    premiumReductionRange: [
      baselineImpact.premium_reduction_low_pct,
      baselineImpact.premium_reduction_high_pct
    ],
    fundingShortfallVsEPTC: Math.max(
      0, metrics.federal_eptc_value_lost_annual_usd - fundingLevelUsd
    ),
    provenanceLabel: atBaseline
      ? "PENNIE_PUBLISHED_POINT"
      : "ASSUMED_LINEAR_SCALING"
  };
};

from functools import wraps
from fastapi import Request, Response
from typing import Callable

def provenance(source: str):
    """Attach a provenance label to the JSON response."""
    def decorator(func: Callable):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            result = await func(*args, **kwargs)
            # FastAPI routes may return dicts, models, or Response objects
            if isinstance(result, (dict, list)):
                if isinstance(result, dict):
                    result["provenanceLabel"] = source
                else:  # list of dicts
                    for item in result:
                        if isinstance(item, dict):
                            item["provenanceLabel"] = source
                return result
            if isinstance(result, Response):
                # Assume JSON payload – inject after body is built
                payload = result.body.decode()
                # Very lightweight injection; for production use a proper model
                # Here we just add the field at the end
                if payload.endswith("}"):
                    payload = payload[:-1] + f', "provenanceLabel":"{source}"}}'
                result.body = payload.encode()
                return result
            return result
        return wrapper
    return decorator
```

#!/usr/bin/env python3
"""
Standalone ETL entry point – can be executed directly or via CI.
"""

import csv
import json
import sys
from src.paths import data_input, data_output

def run() -> None:
    src = data_input()
    dst = data_output()

    if not src.is_file():
        sys.stderr.write(f"[ERROR] Missing source CSV: {src}\\n")
        sys.exit(1)

    out = {}
    with src.open(newline="") as fh:
        reader = csv.DictReader(fh)
        required = {"county", "total_enrollees", "terminations"}
        if not required.issubset(reader.fieldnames or []):
            missing = required - set(reader.fieldnames or [])
            sys.stderr.write(f"[ERROR] CSV missing columns: {missing}\\n")
            sys.exit(1)

        for row in reader:
            county = row["county"].strip()
            try:
                total = int(row["total_enrollees"])
                term = int(row["terminations"])
                rate = round(term / total * 100, 2) if total else 0.0
            except ValueError:
                sys.stderr.write(f"[WARN] Skipping malformed row: {row}\\n")
                continue

            out[county] = {
                "total_enrollees": total,
                "terminations": term,
                "termination_rate_pct": rate,
            }

    dst.parent.mkdir(parents=True, exist_ok=True)
    dst.write_text(json.dumps(out, indent=2))
    print(f"[OK] ETL finished – JSON written to {dst}")

if __name__ == "__main__":
    run()
```

*Key change*: No `Path(__file__).parents[...]` gymnastics; everything goes through `data_input()` / `data_output()`.

### 🎯 Why this “one‑point‑anchor” approach is bulletproof

1. **Zero hard‑coded absolute paths** – the only place a path can leak outside the repo is the optional `$COUNTY_CSV_PATH` env var (explicitly allowed for *inputs*).  
2. **Deterministic output location** – `data_output()` never looks at env vars, so every generated JSON lives under `repo_root/data/quarantine/…`, which is version‑controlled and covered by your provenance ledger.  
3. **Test‑driven guarantee** – the `test_etl_resolves_relative_to_repo_root_not_cwd` test will keep failing if any stray `/home/…` reference sneaks back in.  
4. **Scalable** – any new ETL, seed script, or validation routine can just `from src.paths import data_input, data_output` and be instantly compliant.
