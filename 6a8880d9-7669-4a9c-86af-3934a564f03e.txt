#!/bin/bash
# Quack! Create a PDF↔DOCX plugin for share-tool.

set -euo pipefail

pip install --user pypandoc || { echo "⚠️ Install pypandoc: pip install pypandoc"; exit 1; }

"""
PDF↔DOCX Plugin for Share-Tool
-------------------------------
Converts:
  • PDF → DOCX
  • DOCX → PDF
"""

import pypandoc
from pathlib import Path

class PDFDocxPlugin:
    def __init__(self, config):
        self.config = config
        self.name = "pdf_docx"

    def register_converters(self):
        """Register PDF↔DOCX converters."""
        return {
            "pdf": {
                "to_docx": {
                    "func": self.convert_pdf_to_docx,
                    "output_ext": "docx",
                    "mime_type": "application/vnd.openxmlformats-officedocument.wordprocessingml.document"
                }
            },
            "docx": {
                "to_pdf": {
                    "func": self.convert_docx_to_pdf,
                    "output_ext": "pdf",
                    "mime_type": "application/pdf"
                }
            }
        }

    def convert_pdf_to_docx(self, input_path, output_path):
        """Convert PDF to DOCX."""
        pypandoc.convert_file(
            input_path,
            "docx",
            outputfile=output_path,
            extra_args=["--pdf-engine=pdftoppm"]  # Lightweight engine
        )
        return output_path

    def convert_docx_to_pdf(self, input_path, output_path):
        """Convert DOCX to PDF."""
        pypandoc.convert_file(
            input_path,
            "pdf",
            outputfile=output_path,
            extra_args=["--pdf-engine=wkhtmltopdf"]  # Better formatting
        )
        return output_path

PLUGIN_EOF

# In serve.py, add to SUPPORTED_ROUTES:
from importlib import import_module
from flask import jsonify

@app.route("/plugins")
def list_plugins():
    """List installed plugins and their converters."""
    plugins = []
    for plugin_name in CONFIG.get("plugins", []):
        try:
            plugin_module = import_module(f"plugins.{plugin_name}")
            # Dynamically get the plugin class (e.g., PDFDocxPlugin)
            plugin_class = getattr(plugin_module, f"{plugin_name.replace('_', ' ').title().replace(' ', '')}Plugin")
            plugin = plugin_class(CONFIG)
            plugins.append({
                "name": plugin.name,
                "version": getattr(plugin, "version", "1.0"),
                "converters": plugin.register_converters()
            })
        except Exception as e:
            print(f"⚠️ Plugin {plugin_name} failed to load: {e}")
    return jsonify(plugins)

### **Debugging Common Issues**
If you hit a snag, here’s how to troubleshoot:

#### **1. `pypandoc` Installation Fails**
- **Error:**
  ```
  × This environment is externally managed
  ```
- **Fix:**
  Ensure you’re using the venv:
  ```bash
  source ~/share-tool/plugins/venv_pypandoc/bin/activate
  pip install pypandoc
  deactivate
  ```

#### **2. `pdftoppm` or `wkhtmltopdf` Missing**
- **Error:**
  ```
  pandoc: pdftoppm not found. Please install poppler-utils.
  ```
- **Fix:**
  Install dependencies:
  ```bash
  sudo apt update && sudo apt install poppler-utils wkhtmltopdf
  ```

#### **3. Plugin Not Loading**
- **Error:**
  ```
  ⚠️ Plugin pdf_docx failed to load: ModuleNotFoundError: No module named 'pypandoc'
  ```
- **Fix:**
  Ensure the venv is activated during testing:
  ```python
  import sys
  sys.path.insert(0, "/home/droid/share-tool/plugins/venv_pypandoc/lib/python3.13/site-packages")
  ```

#### **4. Git Tracking Issues**
- **Error:**
  ```
  ⚠️ Git tracking issue
  ```
- **Fix:**
  Manually add the files:
  ```bash
  git add plugins/pdf_docx_plugin.py config.json
  ```
