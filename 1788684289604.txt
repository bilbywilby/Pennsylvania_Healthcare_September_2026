## Corrected `bootstrap/crosswalk_lib.py`

```python
"""Shared crosswalk & filing validation for PA audit pipeline (steps 02, 05, 06)."""
import re
from typing import Any, Dict, List

SERFF_REGEX = re.compile(r"^[A-Z]{4}-\d{9}$")

class CrosswalkLibError(Exception):
    """Structural validation failure in crosswalk or filing data."""

def validate_filings(entries: list) -> list:
    """Validate carrier filings: fail loud on missing fields, dedupe, enforce SERFF format."""
    if not isinstance(entries, list):
        raise CrosswalkError(f"Filings must be a list, got {type(entries).__name__}")
    seen, normalized = set(), []
    for item in entries:
        carrier = (item.get("carrier") or "").strip()
        serff_id = (item.get("serff_id") or "").strip()
        if not carrier or not serff_id:
            raise CrosswalkError(f"Filing missing carrier/serff_id, refusing silent drop: {item!r}")
        if not SERFF_REGEX.match(serff_id):
            raise CrosswalkError(f"Malformed SERFF ID for '{carrier}': {serff_id}")
        key = (carrier, serff_id)
        if key not in seen:
            seen.add(key)
            normalized.append({"carrier": carrier, "serff_id": serff_id, **item})
        else:
            raise CrosswalkError(f"Duplicate (carrier, serff_id): {key}")  # catches the INAC collision
    return normalized

# Crosswalk: same shapes A/B/C, same canonical output, imported by 02, 05, 06.
from typing import List, Any  # placeholder: in practice this file re-exports the canonical normalize_crosswalk
```
