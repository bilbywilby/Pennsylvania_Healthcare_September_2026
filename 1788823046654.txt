const SCRUB_PATTERNS: { id: string; regex: RegExp; label: string }[] = [
  { id: 'SSN',   regex: /\b\d{3}-\d{2}-\d{4}\b/g,                label: 'REDACTED_SSN' },
  { id: 'EMAIL', regex: /\b[a-zA-Z0-9._%+-]+@[a-zA-Z0-9.-]+\.[A-Za-z]{2,}\b/g, label: 'REDACTED_EMAIL' },
  { id: 'PHONE', regex: /\b(?:\+?1[-.\s]?)?\(?\d{3}\)?[-.\s]?\d{3}[-.\s]?\d{4}\b/g, label: 'REDACTED_PHONE' },
  { id: 'NPI',   regex: /\b[12]\d{9}\b/g, label: 'REDACTED_NPI' },
  { id: 'MRN',   regex: /\bMRN\s*[:#-]?\s*[A-Z0-9]{6,12}\b/gi, label: 'REDACTED_MRN' },
  { id: 'DOB',   regex: /\b(?:0?[1-9]|1[0-2])[-/](?:0?[1-9]|[12]\d|3[01])[-/](?:\d{2}|\d{4})\b/g, label: 'REDACTED_DOB' },
];

function toB64(bytes: Uint8Array): string {
  let s = '';
  for (const b of bytes) s += String.fromCharCode(b);
  return btoa(s);
}
function fromB64(b64: string): Uint8Array {
  return Uint8Array.from(atob(b64), (c) => c.charCodeAt(0));
}

let cryptoKey: CryptoKey | null = null;

/** Get or create the per-install HMAC key. Raw material never leaves IndexedDB. */
export async function getHmacKey(): Promise<CryptoKey> {
  if (cryptoKey) return cryptoKey;
  await db.open();
  const existing = await db.secrets.get('hmac-primary');
  if (existing) {
    cryptoKey = await crypto.subtle.importKey(
      'raw', fromB64(existing.hmacRawB64),
      { name: 'HMAC', hash: 'SHA-256' }, false, ['sign'],
    );
    return cryptoKey;
  }
  const raw = crypto.getRandomValues(new Uint8Array(32));
  cryptoKey = await crypto.subtle.importKey(
    'raw', raw, { name: 'HMAC', hash: 'SHA-256' }, false, ['sign'],
  );
  await db.secrets.put({
    key: 'hmac-primary',
    hmacRawB64: toB64(raw),
    createdAt: new Date().toISOString(),
  });
  return cryptoKey;
}

export async function generateHMACPseudonym(value: string): Promise<string> {
  const key = await getHmacKey();
  const sig = await crypto.subtle.sign('HMAC', key, new TextEncoder().encode(value.trim()));
  return Array.from(new Uint8Array(sig))
    .map((b) => b.toString(16).padStart(2, '0')).join('').substring(0, 16);
}

/** Returns ONLY the scrubbed text. No pseudonym-to-plaintext map is ever returned or stored. */
export async function scrubTextPayload(text: string): Promise<string> {
  let out = text;
  for (const p of SCRUB_PATTERNS) {
    out = out.replace(p.regex, (m) =>
      '<' + p.label + ':' + await generateHMACPseudonym(m) + '>');
  }
  return out;
}

export interface AuditParams {
  cptCode: string; pos: number;
  billedAmount: number; allowedAmount: number;
  providerType: 'IN_NETWORK' | 'OUT_OF_NETWORK';
  facilityInNetwork: boolean;
  networkDisclosureProvided: boolean;

function deriveFlag(v: RuleId[], variancePercent: number): AuditFlag {
  if (v.includes(RuleId.NSA_BALANCE_BILL_RISK) || v.includes(RuleId.PA_ACT252_MISSING_DISCLOSURE))
    return 'NSA_REVIEW_REQUIRED';
  if (v.includes(RuleId.PREVENTIVE_COST_SHARING)) return 'COST_SHARING_NOT_ALLOWED';
  if (v.includes(RuleId.PA_SB1071_COINSURANCE_CAP)) return 'COINSURANCE_CAP_EXCEEDED';
  if (variancePercent > 40) return 'CRITICAL_OVERCHARGE';
  if (variancePercent > 15 || v.length > 0) return 'POTENTIAL_AUDIT';
  return 'PASS';
}

export async function executeForensicAudit(params: AuditParams) {
  const violations: RuleId[] = [];
  const benchmark = CPT_BENCHMARKS[params.cptCode] ?? {
    desc: 'Unknown Procedure', fmvLow: 0, fmvHigh: 0,
    medicareBase: params.billedAmount * 0.5, isPreventive: false,
  };

  const contractModifier = params.providerType === 'IN_NETWORK' ? 1.15 : 2.50;
  const fmvTarget = benchmark.medicareBase * contractModifier;
  const variancePercent = ((params.billedAmount - fmvTarget) / fmvTarget) * 100;

  let patientResponsibility = 0;

  if (benchmark.isPreventive) {
    patientResponsibility = 0;
    if (params.billedAmount > 0 &&
        (params.allowedAmount !== params.billedAmount || params.coinsuranceRate > 0)) {
      violations.push(RuleId.PREVENTIVE_COST_SHARING);
    }
  } else {
    if (!params.deductibleMet) {
      patientResponsibility = Math.min(params.billedAmount, params.allowedAmount);
      violations.push(RuleId.DEDUCTIBLE_APPLIED);
    } else {
      patientResponsibility = params.allowedAmount * params.coinsuranceRate;
    }
    if ((params.pos === 11 || params.pos === 22) && params.coinsuranceRate > 0.20) {
      violations.push(RuleId.PA_SB1071_COINSURANCE_CAP);
    }
  }

  if (params.providerType === 'OUT_OF_NETWORK' && params.facilityInNetwork) {
    violations.push(RuleId.NSA_BALANCE_BILL_RISK);
  }
  if (!params.networkDisclosureProvided && params.providerType === 'OUT_OF_NETWORK') {
    violations.push(RuleId.PA_ACT252_MISSING_DISCLOSURE);
  }

  const projectedOop = params.ytdOopSpent + patientResponsibility;
  if (projectedOop > params.planOopMax) {
    const excess = projectedOop - params.planOopMax;
    patientResponsibility = Math.max(0, patientResponsibility - excess);
    violations.push(RuleId.OOP_MAX_EXCEEDED);
  }

  const round2 = (n: number) => Math.round(n * 100) / 100;
  return {
    cptCode: params.cptCode,
    description: benchmark.desc,
    billedAmount: params.billedAmount,
    allowedAmount: params.allowedAmount,
    benchmarkFmv: round2(fmvTarget),
    variancePercent: Math.round(variancePercent * 100) / 100,
    patientResponsibility: round2(patientResponsibility),
    savingsPotential: round2(Math.max(0, params.billedAmount - fmvTarget)),
    auditFlag: deriveFlag(violations, variancePercent),
    ruleViolations: violations,
    displayFlags: violations.map((v) => RULE_DISPLAY[v]),
  };
}

USE_OUTBOX_SYNC_TS = r'''
import { useState, useEffect, useCallback } from 'react';
import { db, OutboxOp } from '../db';

const MAX_RETRIES = 5;
const BASE_DELAY_MS = 2000;
const MAX_DELAY_MS = 60_000;
const POLL_INTERVAL_MS = 5000;

export const useOutboxSync = () => {
  const [online, setOnline] = useState(navigator.onLine);
  const [syncing, setSyncing] = useState(false);
  const [queueSize, setQueueSize] = useState(0);

  const updateQueueSize = useCallback(async () => {
    if (db.isOpen()) setQueueSize(await db.outbox.count());
  }, []);

  useEffect(() => {
    const handleOnline = () => setOnline(true);
    const handleOffline = () => setOnline(false);
    window.addEventListener('online', handleOnline);
    window.addEventListener('offline', handleOffline);
    updateQueueSize();
    return () => {
      window.removeEventListener('online', handleOnline);
      window.removeEventListener('offline', handleOffline);
    };
  }, [updateQueueSize]);

  const enqueueOperation = useCallback(async (
    opType: OutboxOp['opType'], payload: unknown,
  ) => {
    await db.outbox.add({
      opType, payload,
      idempotencyKey: crypto.randomUUID(),
      createdAt: Date.now(),
      retryCount: 0,
      nextAttemptAt: Date.now(),
    });
    await updateQueueSize();
  }, [updateQueueSize]);

  const triggerSync = useCallback(async () => {
    if (!navigator.onLine || syncing) return;
    setSyncing(true);
    try {
      const now = Date.now();
      const due = (await db.outbox.toArray())
        .filter((op) => op.nextAttemptAt <= now)
        .sort((a, b) => (a.seq ?? 0) - (b.seq ?? 0))
        .slice(0, 10);
      for (const op of due) {
        try {
          const res = await fetch('/api/sync', {
            method: 'POST',
            headers: {
              'Content-Type': 'application/json',
              'Idempotency-Key': op.idempotencyKey,
            },
            body: JSON.stringify({ opType: op.opType, payload: op.payload }),
          });
          if (res.status === 409) {
            // Deferred, not failed — another tab holds the lock. Honor Retry-After.
            const ra = Number(res.headers.get('Retry-After')) || 5;
            await db.outbox.update(op.seq!, { nextAttemptAt: Date.now() + ra * 1000 });
            continue;
          }
          if (!res.ok) throw new Error('HTTP ' + res.status);
          await db.outbox.delete(op.seq!);   // delete AFTER confirmed success only
          await markSynced(op);
        } catch {
          if (op.retryCount + 1 >= MAX_RETRIES) continue; // dead-letter locally, warn user
          const delay = Math.min(BASE_DELAY_MS * 2 ** op.retryCount, MAX_DELAY_MS);
          await db.outbox.update(op.seq!, {
            retryCount: op.retryCount + 1,
            lastAttempt: new Date().toISOString(),
            nextAttemptAt: Date.now() + delay,
          });
        }
      }
    } finally {
      await updateQueueSize();
      setSyncing(false);
    }
  }, [syncing, updateQueueSize]);

  // Poll while online — gated by nextAttemptAt, so no hot retry loops.
  useEffect(() => {
    if (!online) return;
    const t = setInterval(() => { if (!syncing) triggerSync(); }, POLL_INTERVAL_MS);
    return () => clearInterval(t);
  }, [online, syncing, triggerSync]);

  return { online, syncing, queueSize, enqueueOperation, triggerSync };
};

async function markSynced(op: OutboxOp): Promise<void> {
  if (op.opType === 'create_claim' || op.opType === 'update_claim') {
    await db.claims.update((op.payload as { id?: string }).id ?? '', { status: 'synced' });
  } else if (op.opType === 'create_report') {
    await db.reports.update((op.payload as { id?: string }).id ?? 'unknown', { status: 'sent' as never });
  }
}

WORKER_TS = r'''
/// <reference types="@cloudflare/workers-types" />

export interface Env {
  DISPATCHER_HMAC_SECRET: string;
  IDEMPOTENCY_KV: KVNamespace;
  INGESTION_URL: string;
}

const MAX_CLOCK_SKEW_MS = 300_000;
const ACK_TTL_SECONDS = 86_400;
const PROCESSING_TTL_SECONDS = 300;

const JSON_HEADERS = { 'Content-Type': 'application/json' };

function json(status: number, body: Record<string, unknown>, extraHeaders: HeadersInit = {}): Response {
  return new Response(JSON.stringify(body), {
    status,
    headers: { ...JSON_HEADERS, ...extraHeaders },
  });
}

function decodeSignature(value: string): Uint8Array | null {
  const raw = value.replace(/^sha256=/i, '').trim();
  if (/^[0-9a-f]{64}$/i.test(raw)) {
    const bytes = new Uint8Array(32);
    for (let i = 0; i < bytes.length; i += 1) {
      bytes[i] = Number.parseInt(raw.slice(i * 2, i * 2 + 2), 16);
    }
    return bytes;
  }
  try {
    const binary = atob(raw);
    return Uint8Array.from(binary, (c) => c.charCodeAt(0));
  } catch {
    return null;
  }
}

async function verifySignature(
  request: Request,
  timestamp: string,
  providedSignature: string,
  bodyText: string,
  secret: string,
): Promise<boolean> {
  const timestampMs = Number(timestamp);
  if (!Number.isFinite(timestampMs) || Math.abs(Date.now() - timestampMs) > MAX_CLOCK_SKEW_MS) {
    return false;
  }
  const signatureBytes = decodeSignature(providedSignature);
  if (!signatureBytes || signatureBytes.length !== 32) return false;

  const url = new URL(request.url);
  // Canonical signing string — the ONLY format; callers must match exactly.
  const canonicalPayload = request.method.toUpperCase() + ' ' + url.pathname + '\n' + timestamp + '\n' + bodyText;

  const cryptoKey = await crypto.subtle.importKey(
    'raw',
    new TextEncoder().encode(secret),
    { name: 'HMAC', hash: 'SHA-256' },
    false,
    ['verify'],
  );
  return crypto.subtle.verify('HMAC', cryptoKey, signatureBytes, new TextEncoder().encode(canonicalPayload));
}

function isValidIdempotencyKey(value: string): boolean {
  return /^[0-9a-f]{8}-[0-9a-f]{4}-[1-5][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i.test(value)
    || /^[a-f0-9]{64}$/i.test(value);
}

interface DispatchRequest {
  opType: string;
  payload: unknown;
}

interface CachedAck {
  status: number;
  body: Record<string, unknown>;
}

export default {
  async fetch(request: Request, env: Env): Promise<Response> {
    if (request.method !== 'POST') {
      return json(405, { error: 'POST only' }, { Allow: 'POST' });
    }

    const idempotencyKey = request.headers.get('Idempotency-Key');
    const signature = request.headers.get('X-Signature-256');
    const timestamp = request.headers.get('X-Timestamp');

    if (!idempotencyKey || !isValidIdempotencyKey(idempotencyKey)) {
      return json(400, { error: 'Missing or malformed Idempotency-Key' });
    }
    if (!signature || !timestamp) {
      return json(401, { error: 'Missing signature headers' });
    }

    const bodyText = await request.text();

    const validSignature = await verifySignature(
      request, timestamp, signature, bodyText, env.DISPATCHER_HMAC_SECRET,
    );
    if (!validSignature) {
      return json(401, { error: 'Unauthorized' });
    }

    let dispatch: DispatchRequest;
    try {
      dispatch = JSON.parse(bodyText) as DispatchRequest;
    } catch {
      return json(400, { error: 'Invalid JSON body' });
    }
    if (typeof dispatch.opType !== 'string' || !Object.prototype.hasOwnProperty.call(dispatch, 'payload')) {
      return json(400, { error: 'Body must contain opType and payload' });
    }

    const ackKey = 'ack:' + idempotencyKey;
    const processingKey = 'processing:' + idempotencyKey;

    // 1. Cached ACK replay — dedupe record is NEVER deleted on success.
    const cached = await env.IDEMPOTENCY_KV.get(ackKey);
    if (cached !== null) {
      try {
        const cachedAck = JSON.parse(cached) as CachedAck;
        return json(cachedAck.status, cachedAck.body, { 'X-Idempotent-Replay': 'true' });
      } catch {
        // Malformed cache entry: treat as unavailable, fall through.
      }
    }

    // 2. Best-effort duplicate signal. KV cannot guarantee locking.
    if ((await env.IDEMPOTENCY_KV.get(processingKey)) !== null) {
      return json(409, { error: 'Request is already being processed', idempotencyKey }, { 'Retry-After': '5' });
    }
    await env.IDEMPOTENCY_KV.put(processingKey, 'processing', { expirationTtl: PROCESSING_TTL_SECONDS });

    try {
      const downstream = await fetch(env.INGESTION_URL, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', 'Idempotency-Key': idempotencyKey },
        body: JSON.stringify({ opType: dispatch.opType, payload: dispatch.payload }),
      });
      if (!downstream.ok) throw new Error('downstream_status_' + downstream.status);

      const ack: CachedAck = {
        status: 202,
        body: { ok: true, status: 'accepted', idempotencyKey },
      };
      await env.IDEMPOTENCY_KV.put(ackKey, JSON.stringify(ack), { expirationTtl: ACK_TTL_SECONDS });
      await env.IDEMPOTENCY_KV.delete(processingKey);
      return json(ack.status, ack.body);
    } catch (error) {
      await env.IDEMPOTENCY_KV.delete(processingKey);
      // Sanitized telemetry: metadata only, zero payload content.
      console.log(JSON.stringify({
        evt: 'dispatch_failure',
        keyPrefix: idempotencyKey.slice(0, 8),
        phase: 'dispatch',
        errKind: error instanceof TypeError ? 'network' : 'downstream_failure',
      }));
      return json(503, { error: 'Upstream unavailable' });
    }
  },
};
'''

def find_root(cli_arg: str | None) -> Path:
    candidates = []
    if cli_arg:
        candidates.append(Path(cli_arg).expanduser())
    home = Path.home()
    candidates.append(home / 'pennsylvania-health-insurance-audit-2026')
    candidates.append(home / 'pa-audit-work')
    for c in candidates:

            return c.resolve()

    for c in candidates:
        print('  -', c, file=sys.stderr)
    print('Re-run with an explicit path: python3 apply_privacy_gate.py /path/to/repo', file=sys.stderr)
    sys.exit(1)


def atomic_write(path: Path, content: str, backup_dir: Path) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    if path.exists():
        rel = path.relative_to(ROOT)
        dest = backup_dir / rel
        dest.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(path, dest)
    fd, tmp_name = tempfile.mkstemp(dir=str(path.parent), suffix='.tmp')
    try:
        with os.fdopen(fd, 'w', encoding='utf-8') as f:
            f.write(content)
        os.replace(tmp_name, path)
    finally:
        if os.path.exists(tmp_name):
            os.unlink(tmp_name)
