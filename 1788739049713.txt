# src/provenance.py
from pydantic import BaseModel, Field
from datetime import date
from typing import Optional, Dict, Any

class Claim(BaseModel):
    claim_id: str = Field(..., description="Unique identifier for the claim")
    value: Any = Field(..., description="The numeric or textual value of the claim")
    source_citation: str = Field(..., description="Human-readable source citation")
    source_url: str = Field(..., description="URL or path to the source document")
    last_verified: date = Field(..., description="ISO date of last verification")

class ProvenanceValidator:
    @staticmethod
    def validate_claim(claim: Dict) -> Claim:
        """Validate a claim against the provenance schema."""
        return Claim(**claim)

    @staticmethod
    def validate_artifact(artifact: Dict, ledger: Dict[str, Claim]) -> bool:
        """Check if all claims in an artifact are in the ledger and current."""
        for claim_id in extract_claim_ids(artifact):
            if claim_id not in ledger:
                raise ValueError(f"Claim {claim_id} not in ledger")
            if ledger[claim_id].last_verified < get_staleness_threshold():
                raise ValueError(f"Claim {claim_id} is stale")
        return True

def extract_claim_ids(artifact: Dict) -> list[str]:

    pass

def get_staleness_threshold() -> date:
    """Return the maximum allowed age for a claim (e.g., 90 days)."""
    from datetime import date, timedelta
    return date.today() - timedelta(days=90)

**Key Features:**
- Uses Pydantic for schema validation.
- Enforces that every claim must have a `claim_id`, `source_citation`, `source_url`, and `last_verified`.
- Provides methods to validate artifacts and check for stale claims.

### 2. **Content Gate in CI**
Add a **pre-commit hook or GitHub Action** to validate all artifacts in `education/` before merging. Example:

```yaml

      - uses: actions/checkout@v4
      - name: Set up Python

          python-version: "3.13"
      - name: Install dependencies
        run: pip install pydantic
      - name: Run provenance validation
        run: python scripts/validate_education_artifacts.py
```

**Script (`scripts/validate_education_artifacts.py`):**
```python
import json
from pathlib import Path
from src.provenance import ProvenanceValidator

LEDGER_PATH = Path("education/verification_log.json")
EDUCATION_ROOT = Path("education")

def main():
    ledger = json.loads(LEDGER_PATH.read_text())
    validator = ProvenanceValidator()
    for artifact_path in EDUCATION_ROOT.rglob("*.json"):
        if artifact_path.name == "verification_log.json":
            continue
        artifact = json.loads(artifact_path.read_text())
        try:
            validator.validate_artifact(artifact, ledger)
        except ValueError as e:
            print(f"❌ Validation failed for {artifact_path}: {e}")
            exit(1)
    print("✅ All education artifacts are valid and current.")

if __name__ == "__main__":
    main()
```

## **Phase 3: Representative Gate (Testing)**
Here’s a **full test suite** for the education gates:

```python
# tests/test_education_gates.py
import json
from pathlib import Path
from src.provenance import ProvenanceValidator

ROOT = Path(__file__).parent.parent / "education"
LEDGER = json.loads((ROOT / "verification_log.json").read_text())

def test_every_numeric_claim_is_sourced():
    for artifact in ROOT.rglob("*.json"):
        if artifact.name == "verification_log.json":
            continue
        data = json.loads(artifact.read_text())
        # Extract claim_ids from the artifact (simplified for example)
        claim_ids = set()
        for value in data.values():
            if isinstance(value, dict) and "claim_id" in value:
                claim_ids.add(value["claim_id"])
        for cid in claim_ids:
            assert cid in LEDGER, f"{artifact.name}: claim '{cid}' not in ledger"

def test_ledger_entries_are_current():
    from datetime import date, timedelta
    staleness_threshold = date.today() - timedelta(days=90)
    for claim_id, entry in LEDGER.items():
        assert "last_verified" in entry, f"{claim_id}: no verification date"
        last_verified = date.fromisoformat(entry["last_verified"])
        assert last_verified >= staleness_threshold, f"{claim_id}: stale (last verified {last_verified})"
```
