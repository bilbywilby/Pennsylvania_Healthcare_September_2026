class PayerModel(Base):
    """Payer registry table"""
    __tablename__ = "payers"
    
    id = Column(Integer, primary_key=True)
    payer_id = Column(String(50), unique=True, nullable=False, index=True)
    payer_name = Column(String(255), nullable=False)
    payer_type = Column(String(100))
    market_share_percent = Column(Numeric(5, 2))
    
    # Contact
    contact_phone = Column(String(20))
    claims_phone = Column(String(20))
    appeals_phone = Column(String(20))
    website = Column(String(255))
    claims_address = Column(Text)
    appeals_address = Column(Text)
    
    # Metadata
    typical_policy_prefix = Column(String(20))
    avg_claim_processing_days = Column(Integer)
    appeal_success_rate = Column(Numeric(5, 4))
    
    # JSON fields
    coverage_regions = Column(JSONB)
    plan_types = Column(ARRAY(String))
    common_denial_reasons = Column(ARRAY(String))
    requires_prior_auth_for = Column(ARRAY(String))
    special_notes = Column(Text)
    
    # Relationships
    contracts = relationship("ContractModel", back_populates="payer")
    claims = relationship("ClaimModel", back_populates="payer")
    
    # Timestamps
    created_at = Column(DateTime, default=datetime.now)
    updated_at = Column(DateTime, default=datetime.now, onupdate=datetime.now)
    
    __table_args__ = (
        Index('idx_payer_name', 'payer_name'),
    )

    PAYER_CACHE = "payer:{payer_id}"

    UPLOAD_SESSION = "upload:{session_id}"
    
    # Real-time analytics

    AUDIT_STATS = "stats:audits:{date}"
    PAYER_PERFORMANCE = "perf:payer:{payer_id}:{month}"
    
    # Rate limiting
    API_RATE_LIMIT = "ratelimit:{user_id}:{endpoint}"
    
    # Locks

class DatabaseQueries:
    """Common database queries"""
    
    @staticmethod
    def get_high_risk_contracts():
        """Get contracts with high risk scores"""
        return """
        SELECT c.contract_id, c.plan_name, p.payer_name, c.overall_risk_score,
               jsonb_array_length(c.deductible_traps) as trap_count
        FROM contracts c
        JOIN payers p ON c.payer_id = p.id
        WHERE c.overall_risk_score > 0.7
        AND c.status = 'flagged'
        ORDER BY c.overall_risk_score DESC
        LIMIT 50;
        """
    
    @staticmethod
    def get_underpaid_claims(min_variance: float = 100.0):
        """Get claims with significant underpayment"""
        return f"""
        SELECT c.claim_id, c.policy_number, p.payer_name,
               c.total_billed, c.total_paid,
               (c.total_billed - c.total_paid) as variance
        FROM claims c
        JOIN payers p ON c.payer_id = p.id
        WHERE c.status = 'paid'
        AND (c.total_billed - COALESCE(c.total_paid, 0)) > {min_variance}
        ORDER BY variance DESC;
        """
    
    @staticmethod
    def get_appeals_due_soon(days: int = 30):
        """Get appeals approaching deadline"""
        return f"""
        SELECT al.appeal_id, al.subject_line, c.claim_id, p.payer_name,
               al.deadline, al.status
        FROM appeal_letters al
        JOIN claims c ON al.claim_id = c.id
        JOIN payers p ON al.payer_id = p.id
        WHERE al.deadline <= NOW() + INTERVAL '{days} days'
        AND al.status IN ('draft', 'ready')
        ORDER BY al.deadline ASC;
        """
    
    @staticmethod
    def get_payer_performance_summary(payer_id: str, start_date: str, end_date: str):
        """Aggregate payer performance"""
        return f"""
        SELECT 
            p.payer_name,
            COUNT(DISTINCT c.id) as total_claims,
            SUM(c.total_billed) as total_billed,
            SUM(c.total_paid) as total_paid,
            AVG(EXTRACT(EPOCH FROM (c.adjudication_date - c.submission_date)) / 86400) as avg_days_to_payment,
            COUNT(CASE WHEN c.status = 'denied' THEN 1 END) as denied_claims,
            COUNT(CASE WHEN c.status = 'denied' THEN 1 END)::float / COUNT(*) as denial_rate
        FROM claims c
        JOIN payers p ON c.payer_id = p.id
        WHERE p.payer_id = '{payer_id}'
        AND c.service_date_start BETWEEN '{start_date}' AND '{end_date}'
        GROUP BY p.payer_name;
        """
