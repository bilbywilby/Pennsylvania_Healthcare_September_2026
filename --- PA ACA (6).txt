# FORENSIC ENGINE - RVU & GPCI

@dataclass
class RVUComponent:
    """Relative Value Unit components for a procedure"""
    work_rvu: Decimal
    practice_expense_rvu: Decimal
    malpractice_rvu: Decimal
    total_rvu: Decimal
    
    def calculate_total(self) -> Decimal:
        return self.work_rvu + self.practice_expense_rvu + self.malpractice_rvu


@dataclass
class GPCIMultiplier:
    """Geographic Practice Cost Index by locality"""
    locality_code: str
    locality_name: str
    work_gpci: Decimal
    practice_expense_gpci: Decimal
    malpractice_gpci: Decimal
    effective_year: int


@dataclass
class ProcedureCode:
    """CPT/HCPCS procedure code with pricing data"""
    code: str
    description: str
    code_type: str  # "CPT" or "HCPCS"
    rvu: RVUComponent
    conversion_factor: Decimal
    
    # Modifiers
    modifiers: List[str] = field(default_factory=list)
    
    # Metadata
    effective_date: datetime = field(default_factory=datetime.now)
    status: str = "active"
    
    def calculate_medicare_rate(self, gpci: GPCIMultiplier) -> Decimal:
        """Calculate Medicare reimbursement rate"""
        adjusted_rvu = (
            (self.rvu.work_rvu * gpci.work_gpci) +
            (self.rvu.practice_expense_rvu * gpci.practice_expense_gpci) +
            (self.rvu.malpractice_rvu * gpci.malpractice_gpci)
        )
        return adjusted_rvu * self.conversion_factor

@dataclass
class LineItem:
    """Individual line item in a claim"""
    line_number: int
    procedure_code: str
    description: str
    units: int
    billed_amount: Decimal
    allowed_amount: Optional[Decimal]
    paid_amount: Optional[Decimal]
    patient_responsibility: Optional[Decimal]
    denial_code: Optional[str] = None
    denial_reason: Optional[str] = None


@dataclass
class Claim:
    """Healthcare claim submission"""
    claim_id: str
    patient_id: str  # Scrubbed identifier
    provider_id: str
    payer_id: str
    
    # Dates
    service_date: datetime
    submission_date: datetime
    adjudication_date: Optional[datetime] = None
    
    # Status
    status: ClaimStatus
    
    # Financial
    line_items: List[LineItem] = field(default_factory=list)
    total_billed: Decimal = Decimal("0.00")
    total_allowed: Optional[Decimal] = None
    total_paid: Optional[Decimal] = None
    patient_responsibility: Optional[Decimal] = None
    
    # Metadata
    diagnosis_codes: List[str] = field(default_factory=list)
    place_of_service: Optional[str] = None
    
    def calculate_totals(self):
        """Calculate claim totals from line items"""
        self.total_billed = sum(item.billed_amount for item in self.line_items)
        if all(item.paid_amount for item in self.line_items):
            self.total_paid = sum(item.paid_amount for item in self.line_items)

@dataclass
class AuditFinding:
    """Individual audit finding"""
    finding_id: str
    finding_type: str  # e.g., "underpayment", "denial_error", "coding_issue"
    severity: RiskLevel
    description: str
    
    # Financial impact
    expected_amount: Decimal
    actual_amount: Decimal
    variance: Decimal
    
    # References
    claim_id: Optional[str] = None
    line_item_ref: Optional[int] = None
    contract_clause_ref: Optional[str] = None
    
    # Evidence
    evidence: List[str] = field(default_factory=list)
    supporting_docs: List[str] = field(default_factory=list)


@dataclass
class AuditReport:
    """Complete forensic audit report"""
    audit_id: str
    claim_id: str
    auditor: str  # System or human identifier
    audit_date: datetime
    
    # Findings
    findings: List[AuditFinding] = field(default_factory=list)
    
    # Summary
    total_variance: Decimal = Decimal("0.00")
    high_priority_count: int = 0
    recommendation: Optional[str] = None
    
    # Validation results
    validation_passed: bool = True
    validation_errors: List[str] = field(default_factory=list)
    
    def calculate_summary(self):
        """Calculate audit summary statistics"""
        self.total_variance = sum(f.variance for f in self.findings)
        self.high_priority_count = sum(
            1 for f in self.findings 
            if f.severity in [RiskLevel.HIGH, RiskLevel.CRITICAL]
        )

@dataclass
class CachedGPCI:
    """Cached GPCI data with TTL"""
    cache_key: str
    locality_code: str
    gpci_data: GPCIMultiplier
    
    # Cache metadata
    cached_at: datetime = field(default_factory=datetime.now)
    ttl_seconds: int = 86400  # 24 hours
    hit_count: int = 0
    
    def is_expired(self) -> bool:
        """Check if cache entry is expired"""
        age = (datetime.now() - self.cached_at).total_seconds()
        return age > self.ttl_seconds

@dataclass
class EOBHeader:
    """EOB document header with member/patient information"""
    eob_id: str
    eob_date: datetime
    
    # Member information
    member_name: str  # Scrubbed
    member_id: str
    policy_number: str
    group_number: Optional[str] = None
    
    # Patient information (if different from member)
    patient_name: str  # Scrubbed
    patient_id: str
    patient_dob: Optional[datetime] = None
    
    # Payer information
    payer_name: str
    payer_id: str
    payer_contact_phone: str
    payer_address: Optional[str] = None
    
    # Plan details
    plan_name: str
    plan_type: str  # "HMO", "PPO", "EPO", "POS"
    network_status: str  # "in_network" or "out_of_network"


@dataclass
class EOBServiceLine:
    """Individual service line on an EOB statement"""
    line_number: int
    
    # Service details
    service_date: datetime
    service_end_date: Optional[datetime] = None
    procedure_code: str  # CPT/HCPCS
    procedure_description: str
    modifiers: List[str] = field(default_factory=list)
    
    # Provider information
    provider_name: str
    provider_npi: Optional[str] = None
    place_of_service: Optional[str] = None
    
    # Financial breakdown (as shown on EOB)
    billed_amount: Decimal  # Provider charges / Amount charged
    allowed_amount: Decimal  # Allowed amount / Contracted rate
    discount_adjustment: Decimal  # Network savings / Adjustment
    
    # Insurance payment breakdown
    plan_paid: Decimal  # Paid by insurer / Insurance paid
    deductible_applied: Decimal  # Applied to deductible
    copay_amount: Decimal  # Copayment
    coinsurance_amount: Decimal  # Coinsurance
    
    # Patient responsibility
    patient_responsibility: Decimal  # Amount you owe / Patient balance
    not_covered_amount: Decimal  # Non-covered charges
    
    # Status codes
    claim_status: str  # "Paid", "Denied", "Partially Paid", "Pending"
    remark_codes: List[str] = field(default_factory=list)
    reason_codes: List[str] = field(default_factory=list)
    denial_reason: Optional[str] = None
    
    def validate_math(self) -> bool:
        """Validate EOB math is correct"""
        # allowed_amount = plan_paid + deductible + copay + coinsurance + not_covered
        calculated_allowed = (
            self.plan_paid + 
            self.deductible_applied + 
            self.copay_amount + 
            self.coinsurance_amount + 
            self.not_covered_amount
        )
        return abs(calculated_allowed - self.allowed_amount) < Decimal("0.01")

@dataclass
class ExplanationOfBenefits:
    """Complete EOB statement (Pennsylvania payers)"""
    eob_id: str
    
    # Header information
    header: EOBHeader
    
    # Service lines
    service_lines: List[EOBServiceLine] = field(default_factory=list)
    
    # Financial summary
    total_billed: Decimal = Decimal("0.00")
    total_allowed: Decimal = Decimal("0.00")
    total_network_savings: Decimal = Decimal("0.00")
    total_plan_paid: Decimal = Decimal("0.00")
    total_patient_responsibility: Decimal = Decimal("0.00")
    
    # Benefit summary
    benefit_summary: Optional[EOBBenefitSummary] = None
    
    # Codes and explanations
    remark_codes: List[EOBRemarkCode] = field(default_factory=list)
    
    # Important notices
    appeal_deadline: Optional[datetime] = None
    appeal_instructions: Optional[str] = None
    
    # Processing metadata
    claim_received_date: Optional[datetime] = None
    claim_processed_date: Optional[datetime] = None
    check_number: Optional[str] = None  # If payment issued
    
    def calculate_totals(self):
        """Calculate EOB financial totals from service lines"""
        self.total_billed = sum(line.billed_amount for line in self.service_lines)
        self.total_allowed = sum(line.allowed_amount for line in self.service_lines)
        self.total_network_savings = sum(line.discount_adjustment for line in self.service_lines)
        self.total_plan_paid = sum(line.plan_paid for line in self.service_lines)
        self.total_patient_responsibility = sum(
            line.patient_responsibility for line in self.service_lines
        )
    
    def get_denied_lines(self) -> List[EOBServiceLine]:
        """Get all denied service lines"""
        return [line for line in self.service_lines if line.claim_status == "Denied"]
    
    def get_variance_lines(self, threshold: Decimal = Decimal("50.00")) -> List[EOBServiceLine]:
        """Get lines with significant payment variance"""
        return [
            line for line in self.service_lines 
            if abs(line.allowed_amount - line.billed_amount) > threshold
        ]
