"""Generates an immutable cryptographic manifest for audit inputs."""

from __future__ import annotations

import datetime
import hashlib
import json
import sys
from pathlib import Path
from typing import Any, Dict, List


def calculate_sha256(file_path: Path) -> str:
    """Calculates SHA-256 checksum in 64KB chunks to handle large datasets efficiently."""
    sha256 = hashlib.sha256()
    with file_path.open("rb") as f:
        while chunk := f.read(65536):
            sha256.update(chunk)
    return sha256.hexdigest()


def generate_manifest(
    data_dir: Path, source_metadata: Dict[str, Dict[str, str]]
) -> Dict[str, Any]:
    """Scans target directory and binds cryptographic checksums with source metadata."""
    manifest: Dict[str, Any] = {
        "generated_at_utc": datetime.datetime.now(datetime.timezone.utc).isoformat(),
        "parser_version": "2.4.0",
        "artifacts": [],
    }

    for file_path in sorted(data_dir.glob("**/*")):
        if file_path.is_file() and not file_path.name.startswith("."):
            rel_path = str(file_path.relative_to(data_dir.parent))
            meta = source_metadata.get(rel_path, {})

            artifact_entry = {
                "file_path": rel_path,
                "sha256": calculate_sha256(file_path),
                "file_size_bytes": file_path.stat().st_size,
                "source_url": meta.get("source_url", "UNSPECIFIED"),
                "retrieval_timestamp": meta.get("retrieved_at", "UNSPECIFIED"),
            }
            manifest["artifacts"].append(artifact_entry)

    return manifest


if __name__ == "__main__":

from __future__ import annotations

import csv
import json
import re
import sys
from pathlib import Path
from typing import Dict, List, Set, Tuple

PA_TOTAL_COUNTIES = 67
VALID_RATING_AREAS = set(range(1, 10))
REQUIRED_COLUMNS = {
    "county_name",
    "fips_code",
    "rating_area",
    "plan_id",
    "effective_date",
    "approved_rate",

# Standard regex for 14-character HIOS Plan ID (e.g., 99812PA0010001)
HIOS_PLAN_ID_REGEX = re.compile(r"^\d{5}[A-Z]{2}\d{7}$")
DATE_REGEX = re.compile(r"^\d{4}-\d{2}-\d{2}$")


def load_canonical_counties(crosswalk_path: Path) -> Dict[str, str]:
    """Loads FIPS to canonical county name mapping."""
    with crosswalk_path.open("r", encoding="utf-8") as f:
        data = json.load(f)
    return {entry["fips"]: entry["canonical_name"] for entry in data["counties"]}


def validate_contract(csv_path: Path, crosswalk_path: Path) -> List[str]:
    """Validates row-level data contract, normalization, and bounds."""
    errors: List[str] = []

    if not csv_path.exists():
        return [f"Fatal: Input file does not exist: {csv_path}"]

    canonical_fips_map = load_canonical_counties(crosswalk_path)
    valid_fips_codes = set(canonical_fips_map.keys())

    with csv_path.open(newline="", encoding="utf-8") as handle:
        reader = csv.DictReader(handle)
        present_columns = set(reader.fieldnames or [])

        missing_cols = REQUIRED_COLUMNS - present_columns
        if missing_cols:
            return [f"Schema Violation: Missing required columns: {sorted(missing_cols)}"]

        seen_keys: Set[Tuple[str, str, str]] = set()
        counties_present: Set[str] = set()

        for line_num, row in enumerate(reader, start=2):
            county = (row.get("county_name") or "").strip()
            fips = (row.get("fips_code") or "").strip()
            rating_area_str = (row.get("rating_area") or "").strip()
            plan_id = (row.get("plan_id") or "").strip()
            effective_date = (row.get("effective_date") or "").strip()
            rate_str = (row.get("approved_rate") or "").strip()

            # 1. FIPS and County Validation
            if fips not in valid_fips_codes:
                errors.append(f"Line {line_num}: Invalid FIPS code '{fips}' for PA.")
            else:
                expected_county = canonical_fips_map[fips]
                if county.casefold() != expected_county.casefold():
                    errors.append(
                        f"Line {line_num}: County name mismatch for FIPS {fips}. "
                        f"Got '{county}', expected '{expected_county}'."
                    )
                counties_present.add(expected_county)

            # 2. Rating Area Bounds Check
            try:
                rating_area = int(rating_area_str)
                if rating_area not in VALID_RATING_AREAS:
                    errors.append(f"Line {line_num}: Rating Area {rating_area} outside allowed bounds [1-9].")
            except ValueError:
                errors.append(f"Line {line_num}: Non-integer rating area '{rating_area_str}'.")

            # 3. Plan ID Standard Format Check

## Critical fixes

### 1. Your crosswalk format does not match the loader

The mapping you supplied is a flat object:

```json
{
  "42001": "Adams",
  "42003": "Allegheny"
}
```

But `load_canonical_counties()` expects this structure:

```json
{
  "counties": [
    {
      "fips": "42001",
      "canonical_name": "Adams"
    }
  ]
}
```

Choose one format. The flat mapping is simpler and less error-prone for this repository. Replace the loader with:

```python
def load_canonical_counties(crosswalk_path: Path) -> dict[str, str]:
    """Load a flat FIPS-to-county mapping."""
    with crosswalk_path.open("r", encoding="utf-8") as handle:
        data = json.load(handle)

    if not isinstance(data, dict):
        raise ValueError("Crosswalk must be a JSON object")

    result: dict[str, str] = {}

    for fips, county in data.items():
        if not isinstance(fips, str) or not re.fullmatch(r"42\d{3}", fips):
            raise ValueError(f"Invalid Pennsylvania FIPS code: {fips!r}")
        if not isinstance(county, str) or not county.strip():
            raise ValueError(f"Invalid county name for FIPS {fips!r}")

        result[fips] = county.strip()

    if len(result) != 67:
        raise ValueError(
            f"Expected 67 Pennsylvania counties, found {len(result)}"
        )

    return result
```

This also makes `PA_TOTAL_COUNTIES` meaningful instead of leaving it unused.

### 2. The source URLs are invalid JSON metadata

These values are Markdown links, not URLs:

```python

```

Use the raw URL:

### 3. The manifest is not actually immutable

A generated JSON file can be edited or regenerated. Call it an **evidence manifest**, then protect it with one or more of:

- committed Git history,
- a signed Git tag,
- a release artifact,
- a detached checksum,
- an append-only audit log.

Add a manifest checksum script:

```bash
mkdir -p audit

cat > audit/hash_manifest.sh <<'BASH'
#!/usr/bin/env bash
set -euo pipefail

manifest="${1:-audit/evidence_manifest.json}"
output="${2:-audit/evidence_manifest.sha256}"

test -f "$manifest"

sha256sum "$manifest" | tee "$output"
BASH

chmod +x audit/hash_manifest.sh
```

Then:

```bash
python3 audit/evidence_manifest.py

```

Do not include the manifest’s own hash inside itself, because that creates a circular dependency.

### 7. Use `Decimal` for rates

from decimal import Decimal, InvalidOperation


try:

2. Precise Contract Validator (audit/contract_validator.py)
Using decimal.Decimal to avoid floating-point rounding errors and checking against the full 67-county JSON object:
#!/usr/bin/env python3
"""Strict schema, decimal-precision, and statutory bounds validator for PA ACA rate datasets."""

from __future__ import annotations

import csv
import datetime
import json
import re
import sys
from decimal import Decimal, InvalidOperation
from pathlib import Path
from typing import Dict, List, Set, Tuple

PA_TOTAL_COUNTIES = 67
VALID_RATING_AREAS = set(range(1, 10))
REQUIRED_COLUMNS = {
    "county_name",
    "fips_code",
    "rating_area",
    "plan_id",
    "effective_date",
    "approved_rate",
}

HIOS_PLAN_ID_REGEX = re.compile(r"^\d{5}[A-Z]{2}\d{7}$")


def load_canonical_crosswalk(crosswalk_path: Path) -> Tuple[Dict[str, str], Dict[str, int]]:
    """Loads FIPS to canonical county name and rating area mappings."""
    with crosswalk_path.open("r", encoding="utf-8") as f:
        data = json.load(f)
    
    counties = data.get("counties", [])
    if len(counties) != PA_TOTAL_COUNTIES:
        raise ValueError(f"Crosswalk validation failure: expected {PA_TOTAL_COUNTIES} counties, found {len(counties)}.")

    fips_to_name = {entry["fips"]: entry["canonical_name"] for entry in counties}
    fips_to_area = {entry["fips"]: entry["rating_area"] for entry in counties}
    return fips_to_name, fips_to_area


def validate_contract(csv_path: Path, crosswalk_path: Path) -> List[str]:
    """Validates row-level data contract, precision, and state-specific logic."""
    errors: List[str] = []

    if not csv_path.exists():
        return [f"Fatal: Target CSV file does not exist: {csv_path}"]

    try:
        fips_to_name, fips_to_area = load_canonical_crosswalk(crosswalk_path)
    except Exception as e:
        return [f"Fatal: Crosswalk file loading failed: {str(e)}"]

    with csv_path.open(newline="", encoding="utf-8") as handle:
        reader = csv.DictReader(handle)
        present_columns = set(reader.fieldnames or [])

        missing_cols = REQUIRED_COLUMNS - present_columns
        if missing_cols:
            return [f"Schema Violation: Missing required columns: {sorted(missing_cols)}"]

        seen_keys: Set[Tuple[str, str, str]] = set()

        for line_num, row in enumerate(reader, start=2):
            county = (row.get("county_name") or "").strip()
            fips = (row.get("fips_code") or "").strip()
            rating_area_str = (row.get("rating_area") or "").strip()
            plan_id = (row.get("plan_id") or "").strip()
            effective_date_str = (row.get("effective_date") or "").strip()
            rate_str = (row.get("approved_rate") or "").strip()

            # FIPS and County Validation
            if fips not in fips_to_name:
                errors.append(f"Line {line_num}: FIPS code '{fips}' is not a valid PA FIPS identifier.")
            else:
                expected_county = fips_to_name[fips]
                if county.casefold() != expected_county.casefold():
                    errors.append(
                        f"Line {line_num}: County name mismatch for FIPS {fips}. "
                        f"Found '{county}', expected '{expected_county}'."
                    )

            # Rating Area Cross-Verification
            try:
                rating_area = int(rating_area_str)
                if rating_area not in VALID_RATING_AREAS:
                    errors.append(f"Line {line_num}: Rating area {rating_area} falls outside legal range [1-9].")
                elif fips in fips_to_area and fips_to_area[fips] != rating_area:
                    errors.append(
                        f"Line {line_num}: Rating area {rating_area} does not match canonical "
                        f"PA Rating Area {fips_to_area[fips]} for FIPS {fips}."
                    )
            except ValueError:
                errors.append(f"Line {line_num}: Rating area '{rating_area_str}' is not an integer.")

            # Plan ID Format
            if not HIOS_PLAN_ID_REGEX.match(plan_id):
                errors.append(f"Line {line_num}: Plan ID '{plan_id}' does not match 14-character HIOS format.")

            # Strict Semantic Date Validation
            try:
                datetime.datetime.strptime(effective_date_str, "%Y-%m-%d")
            except ValueError:
                errors.append(f"Line {line_num}: Date '{effective_date_str}' is not a valid YYYY-MM-DD calendar date.")

            # Decimal Precision Rate Range Check
            try:
                rate = Decimal(rate_str)
                if rate <= Decimal("0.00") or rate > Decimal("10000.00"):
                    errors.append(f"Line {line_num}: Approved rate ${rate} is out of allowable bounds (0, 10000.00].")
            except InvalidOperation:
                errors.append(f"Line {line_num}: Approved rate '{rate_str}' is not a valid decimal number.")

            # Key Uniqueness
            key = (fips, plan_id, effective_date_str)
            if key in seen_keys:
                errors.append(f"Line {line_num}: Duplicate record identified for composite key {key}.")
            seen_keys.add(key)

    return errors


if __name__ == "__main__":
    csv_target = Path(sys.argv[1]) if len(sys.argv) > 1 else Path("data/public/approved_rates_2026.csv")
    crosswalk_target = Path("data/public/pa_county_fips_crosswalk.json")

    validation_errors = validate_contract(csv_target, crosswalk_target)

    if validation_errors:
        print(f"FAILED: {len(validation_errors)} contract violations detected in {csv_target}:")
        for err in validation_errors[:10]:
            print(f"  - {err}")
        if len(validation_errors) > 10:
            print(f"  ... and {len(validation_errors) - 10} additional errors.")
        sys.exit(1)

    print(f"SUCCESS: Data contract verified for {csv_target}.")
    sys.exit(0)

```python
def load_canonical_crosswalk(
    crosswalk_path: Path,
) -> Tuple[Dict[str, str], Dict[str, int]]:
    """Load and strictly validate the Pennsylvania county crosswalk."""
    with crosswalk_path.open("r", encoding="utf-8") as handle:
        data = json.load(handle)

    if data.get("state_fips") != "42":
        raise ValueError("Crosswalk state_fips must be '42'")

    if data.get("state_abbr") != "PA":
        raise ValueError("Crosswalk state_abbr must be 'PA'")

    counties = data.get("counties")

    if not isinstance(counties, list):
        raise ValueError("Crosswalk counties must be a list")

    if len(counties) != PA_TOTAL_COUNTIES:
        raise ValueError(
            f"Expected {PA_TOTAL_COUNTIES} counties, found {len(counties)}"
        )

    fips_to_name: Dict[str, str] = {}
    fips_to_area: Dict[str, int] = {}

    for index, entry in enumerate(counties, start=1):
        if not isinstance(entry, dict):
            raise ValueError(f"County entry {index} must be an object")

        fips = entry.get("fips")
        county_name = entry.get("canonical_name")
        rating_area = entry.get("rating_area")

        if not isinstance(fips, str) or not re.fullmatch(r"42\d{3}", fips):
            raise ValueError(f"Invalid Pennsylvania FIPS at entry {index}: {fips!r}")

        if fips in fips_to_name:
            raise ValueError(f"Duplicate FIPS code: {fips}")

        if not isinstance(county_name, str) or not county_name.strip():
            raise ValueError(f"Invalid county name for {fips}")

        if (
            isinstance(rating_area, bool)
            or not isinstance(rating_area, int)
            or rating_area not in VALID_RATING_AREAS
        ):
            raise ValueError(
                f"Invalid rating area for {fips}: {rating_area!r}"
            )

        fips_to_name[fips] = county_name.strip()
        fips_to_area[fips] = rating_area

    expected_fips = {f"42{code:03d}" for code in range(1, 134, 2)}

    if set(fips_to_name) != expected_fips:
        missing = sorted(expected_fips - set(fips_to_name))
        unexpected = sorted(set(fips_to_name) - expected_fips)
        raise ValueError(
            f"FIPS set mismatch; missing={missing}, unexpected={unexpected}"
        )

    return fips_to_name, fips_to_area
```
