    # Step 3: Fetch dump via w3m
    # Note: w3m proxy config varies; usually handled via ~/.w3m/config or env
    w3m -dump -U "$UA" "$url" > "$DUMP_FILE" 2>/dev/null
    rc=$?

    # Step 4: Validity Guards
    # Guard A: Size Check (5KB threshold)
    FILE_SIZE=$(wc -c < "$DUMP_FILE")

        echo -e "${R}⚠️ TOO SMALL ($FILE_SIZE bytes)${NC}"
        echo "$(date +%T),$rc,TOO_SMALL" >> "$LOG"
        continue
    fi

    # Guard B: Content Check
    if ! grep -qiE "<html>|<title|body" "$DUMP_FILE"; then
        echo -e "${R}⚠️ INVALID CONTENT${NC}"
        echo "$(date +%T),$rc,INVALID_CONTENT" >> "$LOG"
        continue
    fi

    # Step 5 & 6: Clean and Compute SHA256
    # We pipe the dump through the Python cleaner before hashing to remove noise
    NEW_HASH=$(cat "$DUMP_FILE" | python3 "$CLEANER" | sha256sum | awk '{print $1}')
    
    if [[ -f "$HASH_FILE" ]]; then
        OLD_HASH=$(cat "$HASH_FILE")
        if [[ "$NEW_HASH" != "$OLD_HASH" ]]; then
            # Step 7: Diff Detected
            TIMESTAMP=$(date +%Y%m%d_%H%M%S)
            DIFF_FILE="$VAULT/diff_${tid}_${TIMESTAMP}.txt"
            # Simple diff of the cleaned versions
            diff <(cat "$DUMP_FILE" | python3 "$CLEANER") <(cat "$DUMP_FILE" | python3 "$CLEANER" --old) > "$DIFF_FILE" 2>/dev/null || echo "Change detected in state." > "$DIFF_FILE"
            
            echo -e "${G}CHANGE_DETECTED${NC}"
            echo "$(date +%T),$rc,CHANGE_DETECTED" >> "$LOG"
            
            # Step 8: Update baseline
            echo "$NEW_HASH" > "$HASH_FILE"
        else
            echo -e "${NC}NO_CHANGE${NC}"
            echo "$(date +%T),$rc,STABLE" >> "$LOG"
        fi
    else
        # Initialize baseline if not exists
        echo "$NEW_HASH" > "$HASH_FILE"
        echo -e "${G}BASELINE_SET${NC}"
        echo "$(date +%T),$rc,INIT" >> "$LOG"
    fi
done
