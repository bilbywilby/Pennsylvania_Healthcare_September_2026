```python
"""
PENNIE / PID COVERAGE AREAS CRAWLER
Parses public 2026 county listings from PID (no SERFF, no RTKL needed)
"""
import logging
import re
from urllib.parse import urljoin
import requests
from bs4 import BeautifulSoup
from config import REQUEST_DELAY_SEC, USER_AGENT

class PennieCoverageParser:
    """Parses PID coverage-areas pages (individual + small group)"""
    
    BASE_URL = "www.pa.gov/agencies/insurance/posted-filings-reports-company-orders/product-and-rate-filings/aca-health-rate-filings"
    INDIVIDUAL_URL = BASE_URL + "/coverage-areas-individual-market"
    SMALL_GROUP_URL = BASE_URL + "/coverage-areas-small-group-market"  # exists per earlier search

    def __init__(self):
        self.session = requests.Session()
        self.session.headers.update({"User-Agent": USER_AGENT})

    def get_coverage_dict(self, market: str = "individual") -> dict[str, list[str]]:
        """Returns {carrier_name_normalized: [counties...]}"""
        url = self.INDIVIDUAL_URL if market == "individual" else self.SMALL_GROUP_URL
        time.sleep(REQUEST_DELAY_SEC)
        html = self.session.get(url, timeout=30).text
        soup = BeautifulSoup(html, "html.parser")

        coverage = {}
        # Each carrier is a heading + paragraph with county list
        sections = soup.find_all(["h2", "h3", "p"])
        current_carrier = None

        for el in sections:
            text = el.get_text(strip=True)
            if el.name in ("h2", "h3") and any(kw in text.lower() for kw in ["inc", "health plan", "assurance", "partners"]):
                current_carrier = self._normalize_carrier(text)
            elif current_carrier and "covers the following" in text.lower():
                counties = re.findall(r'([A-Za-z]+(?:\s+[A-Za-z]+)*)', text.split(":", 1)[-1])
                counties = [c.strip() for c in counties if c.strip() and c not in ("Pennsylvania", "counties")]
                coverage[current_carrier] = counties

        logging.info(f"Parsed {len(coverage)} carriers from {market} coverage page")
        return coverage

    @staticmethod
    def _normalize_carrier(name: str) -> str:
        return re.sub(r'\s+', ' ', name).strip().lower().replace("inc.", "inc").replace("llc", "")
```

```python
"""
PARSes PENNIE PUBLIC AFFORDABILITY PDFs
Premium changes + terminations by county (2026)

import logging
from pathlib import Path
import pdfplumber
import requests
import time
from config import REQUEST_DELAY_SEC

class PennieAffordabilityParser:
    PDF_URLS = {
        "premium_pct": "pennie.com/wp-content/uploads/2026/02/Net-Premium-Change-in-Dollar-Percentage-for-Pre-Renewal-Run-02.pdf",
        "premium_dollar": "pennie.com/wp-content/uploads/2026/02/Net-Premium-Change-in-Dollar-Percentage-for-Pre-Renewal-Run-01.pdf",
        "terminations": "pennie.com/wp-content/uploads/2026/02/Open-Enrollment-Comparison-2021-2026-43-Terminations-by-County-OE-2026-1.pdf"
    }

    def fetch_and_parse(self) -> dict:
        data = {}
        for key, url in self.PDF_URLS.items():
            time.sleep(REQUEST_DELAY_SEC)
            resp = requests.get(url, timeout=30)
            resp.raise_for_status()
            with pdfplumber.open(io.BytesIO(resp.content)) as pdf:
                text = "\n".join(page.extract_text() or "" for page in pdf.pages)
                data[key] = self._extract_county_values(text, key)
        return data

    @staticmethod
    def _extract_county_values(text: str, key: str) -> dict[str, float]:
        # Heuristic – PDFs contain lines like "Lehigh County: 205%" or tables
        county_map = {}
        lines = text.split("\n")
        for line in lines:
            match = re.search(r'([A-Za-z]+(?:\s+[A-Za-z]+)*)\s*[:\-]?\s*([\d.]+)', line)
            if match:
                county = match.group(1).strip().title()
                val = float(match.group(2))
                if "lehigh" in county.lower():
                    logging.info(f"Pennie Lehigh {key}: {val}")
                county_map[county] = val
        return county_map
```

"""
```python
Pennie 2026 affordability parser.
"""
import logging, io, re, hashlib, time
import pdfplumber, requests
from config import REQUEST_DELAY_SEC

class PennieAffordabilityParser:
    PDF_URLS = {
        "pct": "pennie.com/wp-content/uploads/2026/02/Net-Premium-Change-in-Dollar-Percentage-for-Pre-Renewal-Run-02.pdf",
        "dollar": "pennie.com/wp-content/uploads/2026/02/Net-Premium-Change-in-Dollar-Percentage-for-Pre-Renewal-Run-01.pdf",
        "term": "pennie.com/wp-content/uploads/2026/02/Open-Enrollment-Comparison-2021-2026-43-Terminations-by-County-OE-2026-1.pdf"
    }

    def fetch_and_parse(self):
        """Fetch + parse all three PDFs."""
        data = {}
        h = hashlib.sha256()
        for k, u in self.PDF_URLS.items():
            time.sleep(REQUEST_DELAY_SEC)
            try:
                r = requests.get(u, timeout=30)
                r.raise_for_status()
                b = r.content
                h.update(b)
                with pdfplumber.open(io.BytesIO(b)) as pdf:
                    txt = "\n".join(p.extract_text() or "" for p in pdf.pages if p.extract_text())
                    data[k] = self._parse(txt, k)
            except Exception as e:
                logging.error(f"Pennie {k} failed: {e}")
                data[k] = {}
        data["hash"] = h.hexdigest()
        return data

    @staticmethod
    def _parse(text, key):
        """County → value map."""
        m = {}
        for line in text.split("\n"):
            match = re.search(r"([A-Za-z\s]+?)\s*[:\-]?\s*([\d.,]+)%?", line.strip())
            if match:
                c = match.group(1).strip().title()
                v = match.group(2).replace(",", "")
                try:
                    val = float(v)
                    if "term" in key: val = int(val)
                    m[c] = val
                except ValueError:
                    pass
        return m
```

"""
Pennie 2026 affordability parser.
"""
import logging
import io
import re
import hashlib
import time
import pdfplumber
import requests
from config import REQUEST_DELAY_SEC, PDF_URLS

class PennieAffordabilityParser:
    def fetch_and_parse(self):
        """Fetch + parse all three PDFs."""
        data = {}
        h = hashlib.sha256()
        for k, u in PDF_URLS.items():
            time.sleep(REQUEST_DELAY_SEC)
            try:
                r = requests.get(u, timeout=30)
                r.raise_for_status()
                b = r.content
                h.update(b)
                with pdfplumber.open(io.BytesIO(b)) as pdf:
                    txt = "\n".join(p.extract_text() or "" for p in pdf.pages if p.extract_text())
                    data[k] = self._parse(txt, k)
            except Exception as e:
                logging.error(f"Pennie {k} failed: {e}")
                data[k] = {}
        data["hash"] = h.hexdigest()
        return data

    @staticmethod
    def _parse(text, key):
        """County → value map."""
        m = {}
        for line in text.split("\n"):
            match = re.search(r"([A-Za-z\s]+?)\s*[:\-]?\s*([\d.,]+)%?", line.strip())
            if match:
                c = match.group(1).strip().title()
                v = match.group(2).replace(",", "")
                try:
                    val = float(v)
                    if "term" in key: val = int(val)
                    m[c] = val
                except ValueError:
                    pass
        return m
